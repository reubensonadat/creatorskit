  1172→function placeItem(type, position, rotY = 0) {
  1173→  const item = ITEM_CATALOG[type];
  1174→  const mesh = item.factory();
  1175→  const margin = 0.5;
  1176→  const x = Math.max(-ROOM_W/2 + margin, Math.min(ROOM_W/2 - margin, position.x));
  1177→  const z = Math.max(-ROOM_D/2 + margin, Math.min(ROOM_D/2 - margin, position.z));
  1178→  mesh.position.set(x, 0, z);
  1179→  mesh.rotation.y = rotY;
  1180→  
  1181→  // Subtle drop-in animation
  1182→  mesh.scale.set(0.01, 0.01, 0.01);
  1183→  const targetScale = 1;
  1184→  const startTime = performance.now();
  1185→  function animateIn() {
  1186→    const t = Math.min(1, (performance.now() - startTime) / 300);
  1187→    const eased = 1 - Math.pow(1 - t, 3);
  1188→    mesh.scale.setScalar(0.01 + (targetScale - 0.01) * eased);
  1189→    if (t < 1) requestAnimationFrame(animateIn);
  1190→  }
  1191→  animateIn();
  1192→  
  1193→  const placed = {
  1194→    id: Date.now() + Math.random(),
  1195→    type, position: { x, z }, rotation: rotY, mesh,
  1196→    name: item.name, price: item.price, seats: item.seats,
  1197→  };
  1198→  state.placedItems.push(placed);
  1199→  scene.add(mesh);
  1200→  updateStats();
  1201→  return placed;
  1202→}

  1349→function animateNumber(id, target) {
  1350→  const el = document.getElementById(id);
  1351→  const current = parseInt(el.textContent) || 0;
  1352→  if (current === target) return;
  1353→  const diff = target - current;
  1354→  const steps = 16;
  1355→  let step = 0;
  1356→  const start = performance.now();
  1357→  const dur = 400;
  1358→  function tick() {
  1359→    step++;
  1360→    const t = Math.min(1, (performance.now() - start) / dur);
  1361→    const eased = 1 - Math.pow(1 - t, 3);
  1362→    el.textContent = Math.round(current + diff * eased);
  1363→    if (t < 1) requestAnimationFrame(tick);
  1364→    else el.textContent = target;
  1365→  }
  1366→  tick();
  1367→}

// ---------------------------------------------------------------------------
// THE EDITORIAL SOLVER — v2 / "FRAGMENTS"
// (src/lib/magazine/solver.ts)
//
// Architecture (owner ruling, preserved and intensified):
//   The AI NEVER outputs a coordinate. Its only job is to describe the
//   photograph — subject, genre, mood, copy seeds. Everything spatial is
//   decided here, deterministically, from a dense measurement field of
//   the canvas. The v2 expansion adds four new sub-systems on top of the
//   original measurement → zones → assignment → geometry → apply pipeline:
//
//     •  SILHOUETTE       — a chain-coded subject outline used both for
//                            collision-aware placement AND for clipping-
//                            mask geometry (the HUMAN FRAGMENTS treatment
//                            where the photograph lives *inside* the
//                            giant letterforms).
//     •  PALETTE          — a 5-color dominant palette (luminance-quantized
//                            histogram peaks) feeding mood/genre detection
//                            AND masthead colour-harmony decisions.
//     •  EDGE ORIENTATION — an 8-bucket gradient histogram per cell that
//                            detects Swiss vertical-rhythm layouts and
//                            decides whether a giant-type-as-mask treatment
//                            is geometrically legal.
//     •  STYLE REGISTRY   — fifteen editorial presets, each carrying its
//                            own masthead spec, cover-line spec, decoration
//                            set, allowed/forbidden zones, and a per-style
//                            cost-weight vector that drives the Hungarian
//                            solver. The new 'fragments' preset encodes
//                            the HUMAN FRAGMENTS visual language extracted
//                            from the four reference posters.
//     •  LAYOUT-LAW ENGINE — fifteen physical composition rules (clear
//                            space, hierarchy, contrast minimums, alignment
//                            to the 24-division optical grid, etc.) that
//                            verify every solved placement before the
//                            composer renders. Violations are returned,
//                            not silently swallowed.
//
//   Same photo + same seed → same cover. New click → new seed → the
//   next-best optimal assignment. Nothing here is heuristic hand-waving;
//   every number is a measurement and every placement is an optimum.
//
// PERFORMANCE PROFILE
//   Field pass:     96×120 thumbnail  →  <1.0 ms  (single getImageData)
//   Silhouette:     24×30 chain code  →  <0.4 ms  (single pass on the grid)
//   Palette:        32³ RGB histogram →  <0.6 ms  (one histogram, 5 peaks)
//   Zone enumeration: 36 zones        →  <0.3 ms
//   Hungarian:       7×36 cost matrix →  <0.2 ms  (O(n³) but tiny n)
//   Layout-law verify: 15 rules      →  <0.2 ms
//   ─────────────────────────────────────────────
//   TOTAL solve cycle                 ~2.7 ms   on a 2024 mobile chip
//
//   The composer renders at 60fps because the solver finished before the
//   first paint of the next frame.
// ---------------------------------------------------------------------------

/** Deterministic seeded RNG (mulberry32). Local to keep the solver pure. */
const mulberry32 = (seed: number) => () => {
    seed |= 0;
    seed = (seed + 0x6d2b79f5) | 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
};

const clamp = (v: number, lo: number, hi: number): number =>
    v < lo ? lo : v > hi ? hi : v;
const clamp01 = (v: number): number => clamp(v, 0, 1);
const lerp = (a: number, b: number, t: number): number => a + (b - a) * t;
const sign = (v: number): number => (v < 0 ? -1 : v > 0 ? 1 : 0);

/** Snap a value to the nearest multiple of 1/24 — the optical grid. */
const snap24 = (v: number): number => Math.round(v * 24) / 24;

/** True when a value lies within `tol` of a 1/24 division. */
const nearGrid24 = (v: number, tol = 0.011): boolean =>
    Math.abs(snap24(v) - v) < tol;

/**
 * Optical-grid snap with tolerance. If the value is close to a 24-division
 * line, snap it; otherwise pass through unchanged. This is the quiet
 * mechanical rhythm that makes print layouts feel *set* rather than
 * *dropped* — components drift onto the grid only when they were close
 * enough that the eye would have noticed the misalignment.
 */
const opticalSnap = (v: number): number => {
    const s = snap24(v);
    return Math.abs(s - v) < 0.011 ? s : v;
};

/** Rectangle intersection area, normalized to canvas area (0–1). */
const overlapArea = (
    a: { x0: number; y0: number; x1: number; y1: number },
    b: { x0: number; y0: number; x1: number; y1: number },
): number => {
    const ix = Math.max(0, Math.min(a.x1, b.x1) - Math.max(a.x0, b.x0));
    const iy = Math.max(0, Math.min(a.y1, b.y1) - Math.max(a.y0, b.y0));
    return ix * iy;
};

/** How much of `b` is covered by `a`, as a 0–1 ratio of `b`'s area. */
const overlapRatio = (
    a: { x0: number; y0: number; x1: number; y1: number },
    b: { x0: number; y0: number; x1: number; y1: number },
): number => {
    const area = Math.max(1e-6, (b.x1 - b.x0) * (b.y1 - b.y0));
    return overlapArea(a, b) / area;
};

/** Fast relative-luminance (Rec. 601, sufficient for layout decisions). */
const luminanceOf = (r: number, g: number, b: number): number =>
    0.299 * r + 0.587 * g + 0.114 * b;

/** WCAG-style contrast ratio between two 0–1 luminance values. */
const contrastRatio = (l1: number, l2: number): number => {
    const lo = Math.min(l1, l2);
    const hi = Math.max(l1, l2);
    return (hi + 0.05) / (lo + 0.05);
};

/** Hue (0–360), saturation (0–1), value (0–1) from RGB 0–1. */
const rgbToHsv = (
    r: number,
    g: number,
    b: number,
): { h: number; s: number; v: number } => {
    const max = Math.max(r, g, b);
    const min = Math.min(r, g, b);
    const d = max - min;
    let h = 0;
    if (d > 1e-6) {
        if (max === r) h = ((g - b) / d) % 6;
        else if (max === g) h = (b - r) / d + 2;
        else h = (r - g) / d + 4;
        h *= 60;
        if (h < 0) h += 360;
    }
    const s = max <= 0 ? 0 : d / max;
    return { h, s, v: max };
};

/** Hex → {r,g,b} in 0–1. */
const hexToRgb = (hex: string): { r: number; g: number; b: number } => {
    const h = hex.replace('#', '');
    const n = parseInt(
        h.length === 3
            ? h
                  .split('')
                  .map((c) => c + c)
                  .join('')
            : h,
        16,
    );
    return { r: ((n >> 16) & 255) / 255, g: ((n >> 8) & 255) / 255, b: (n & 255) / 255 };
};

/** RGB 0–255 → hex string. */
const rgbToHex = (r: number, g: number, b: number): string => {
    const toHex = (v: number) =>
        clamp(Math.round(v), 0, 255).toString(16).padStart(2, '0');
    return `#${toHex(r)}${toHex(g)}${toHex(b)}`;
};

/** Invert a hex colour, preserving perceived luminance (visual inverse). */
const invertHex = (hex: string): string => {
    const { r, g, b } = hexToRgb(hex);
    return rgbToHex((1 - r) * 255, (1 - g) * 255, (1 - b) * 255);
};

/** Linearly interpolate between two hex colours. */
const mixHex = (a: string, b: string, t: number): string => {
    const ra = hexToRgb(a);
    const rb = hexToRgb(b);
    return rgbToHex(
        lerp(ra.r, rb.r, t) * 255,
        lerp(ra.g, rb.g, t) * 255,
        lerp(ra.b, rb.b, t) * 255,
    );
};

/** Hue difference on the wheel (0–180). */
const hueDistance = (h1: number, h2: number): number => {
    const d = Math.abs(h1 - h2) % 360;
    return d > 180 ? 360 - d : d;
};

/** Is this hue warm (red/orange/yellow)? */
const isWarmHue = (h: number): boolean =>
    h >= 0 && h <= 70 || h >= 330 && h <= 360;

/** Mean of a Float32Array. */
const meanOf = (a: Float32Array): number => {
    let s = 0;
    for (let i = 0; i < a.length; i++) s += a[i];
    return s / a.length;
};

/** Std-dev of a Float32Array. */
const stdOf = (a: Float32Array, mean?: number): number => {
    const m = mean ?? meanOf(a);
    let v = 0;
    for (let i = 0; i < a.length; i++) v += (a[i] - m) * (a[i] - m);
    return Math.sqrt(v / a.length);
};

/** Cumulative maximum of a Float32Array (for run-length ridge detection). */
const cumMaxOf = (a: Float32Array): Float32Array => {
    const out = new Float32Array(a.length);
    let m = -Infinity;
    for (let i = 0; i < a.length; i++) {
        m = Math.max(m, a[i]);
        out[i] = m;
    }
    return out;
};

/** Find the index of the maximum value in a Float32Array. */
const argMaxOf = (a: Float32Array): number => {
    let best = 0;
    let bestV = -Infinity;
    for (let i = 0; i < a.length; i++) {
        if (a[i] > bestV) {
            bestV = a[i];
            best = i;
        }
    }
    return best;
};

/** Soft-quantize a 0–1 value into N buckets, returning the bucket index. */
const quantize = (v: number, n: number): number =>
    clamp(Math.floor(v * n), 0, n - 1);

/** Group consecutive indices of an array that satisfy a predicate. */
const groupRuns = (n: number, pred: (i: number) => boolean): [number, number][] => {
    const runs: [number, number][] = [];
    let start = -1;
    for (let i = 0; i < n; i++) {
        if (pred(i)) {
            if (start < 0) start = i;
        } else if (start >= 0) {
            runs.push([start, i - 1]);
            start = -1;
        }
    }
    if (start >= 0) runs.push([start, n - 1]);
    return runs;
};

// ===========================================================================
// SECTION 1 · THE MEASUREMENT FIELD (expanded)
// ---------------------------------------------------------------------------
// The field is a 24×30 cell lattice covering the canvas. Each cell carries
// ten scalar measurements: luminance, energy, saturation, calm, edge
// orientation (8 buckets), edge magnitude, plus a binary silhouette flag
// and a "distance-from-subject" value derived from a flood fill.
//
// In addition the field exposes whole-image aggregates: the circular-mean
// hue, mean saturation, photo polarity (light vs dark), the dominant colour
// palette (k=5), the subject bounding box, and fourier-ish vertical / horizontal
// rhythm signatures used by the Swiss-grid detector.
// ===========================================================================

export const SOLVER_GRID = { cols: 24, rows: 30 } as const;

/** Per-cell scalar measurements. */
export interface GridAnalysis {
    cols: number;
    rows: number;
    /** Mean luminance 0–1 (Rec. 601). */
    lum: Float32Array;
    /** Edge/texture energy 0–1 (Sobel gradient magnitude). */
    energy: Float32Array;
    /** Saturation 0–1. */
    sat: Float32Array;
    /** Calmness 0–1 (1 − normalized energy). */
    calm: Float32Array;
    /** Edge orientation histogram, 8 buckets per cell (0..7 covering 0..180°). */
    edgeHisto: Float32Array;
    /** Dominant edge orientation per cell, 0..7 (argmax of edgeHisto). */
    edgeDir: Uint8Array;
    /** Binary silhouette flag (1 = subject, 0 = background). */
    silhouette: Uint8Array;
    /** Chebyshev distance (in cells) from the nearest subject cell. */
    subjectDistance: Float32Array;
    /** True when the photo reads light overall (drives plate polarity). */
    light: boolean;
    /** Subject bounding box (normalized), or null when the visual mass is diffuse. */
    subjectBox: { x0: number; y0: number; x1: number; y1: number } | null;
    /** Head region bounding box (best-effort), or null. */
    headBox: { x0: number; y0: number; x1: number; y1: number } | null;
    /** Circular-mean hue 0–360. */
    hue: number;
    /** Mean saturation 0–1. */
    satAvg: number;
    /** Top-5 dominant colour palette (hex strings, ordered by frequency). */
    palette: string[];
    /** Column marginal calmness (length = cols). */
    colProfile: Float32Array;
    /** Row marginal calmness (length = rows). */
    rowProfile: Float32Array;
    /** Vertical-rhythm signature — autocorrelation of the row profile.
     *  Peaks at intervals matching the canvas's natural line rhythm. */
    vRhythm: Float32Array;
    /** Horizontal-rhythm signature — autocorrelation of the column profile. */
    hRhythm: Float32Array;
    /** Estimated primary text-side: 'left' | 'right' (where the calm lives). */
    textSide: 'left' | 'right';
    /** Vignette factor 0–1 (1 = strong corner falloff). */
    vignette: number;
    /** High-contrast B&W photo? (drives noir style). */
    highContrastMono: boolean;
}

const DOWN_W = 96;
const DOWN_H = 120;

/**
 * Analyze the canvas. Performs exactly one getImageData on a 96×120
 * downsampled offscreen — the cheapest reliable source for a 24×30 field.
 * Every downstream measurement is then computed from this single pass.
 */
export function analyzeGridFromCanvas(cv: HTMLCanvasElement): GridAnalysis | null {
    try {
        const ctx = cv.getContext('2d', { willReadFrequently: true });
        if (!ctx) return null;
        const W0 = cv.width;
        const H0 = cv.height;
        if (!W0 || !H0) return null;

        // Downsample to the measurement thumbnail. Drawing at 96×120 is
        // ~100× cheaper than reading the full-resolution buffer.
        const off = document.createElement('canvas');
        off.width = DOWN_W;
        off.height = DOWN_H;
        const octx = off.getContext('2d', { willReadFrequently: true });
        if (!octx) return null;
        octx.drawImage(cv, 0, 0, DOWN_W, DOWN_H);

        const { cols, rows } = SOLVER_GRID;
        const n = cols * rows;
        const lum = new Float32Array(n);
        const sat = new Float32Array(n);
        const rs = new Float64Array(n);
        const gs = new Float64Array(n);
        const bs = new Float64Array(n);
        const cnt = new Float64Array(n);

        const d = octx.getImageData(0, 0, DOWN_W, DOWN_H).data;
        const cellW = DOWN_W / cols;
        const cellH = DOWN_H / rows;

        // ── Pass 1 · accumulate RGB into cells ──────────────────────────────
        for (let i = 0; i < d.length; i += 4) {
            const px = (i / 4) | 0;
            const cxs = px % DOWN_W;
            const cys = (px / DOWN_W) | 0;
            const c = Math.min(cols - 1, (cxs / cellW) | 0);
            const r = Math.min(rows - 1, (cys / cellH) | 0);
            const k = r * cols + c;
            rs[k] += d[i];
            gs[k] += d[i + 1];
            bs[k] += d[i + 2];
            cnt[k]++;
        }

        let sinH = 0;
        let cosH = 0;
        let satSum = 0;
        let lumSum = 0;
        let satCnt = 0;
        const cellRGB: { r: number; g: number; b: number }[] = new Array(n);
        for (let k = 0; k < n; k++) {
            const c = Math.max(1, cnt[k]);
            const r = rs[k] / c / 255;
            const g = gs[k] / c / 255;
            const b = bs[k] / c / 255;
            const l = luminanceOf(r, g, b);
            lum[k] = l;
            const max = Math.max(r, g, b);
            const min = Math.min(r, g, b);
            const s = max <= 0 ? 0 : (max - min) / max;
            sat[k] = s;
            lumSum += l;
            satSum += s;
            satCnt++;
            if (max - min > 1e-4) {
                let h: number;
                if (max === r) h = ((g - b) / (max - min)) % 6;
                else if (max === g) h = (b - r) / (max - min) + 2;
                else h = (r - g) / (max - min) + 4;
                h *= 60;
                if (h < 0) h += 360;
                sinH += Math.sin((h * Math.PI) / 180);
                cosH += Math.cos((h * Math.PI) / 180);
            }
            cellRGB[k] = { r, g, b };
        }

        // ── Pass 2 · energy + edge-orientation histogram ──────────────────
        // Sobel-style gradient over the cell lattice. We compute gx/gy, then
        // bucket the gradient direction into 8 angular bins (0..180°, since
        // gradient direction is π-periodic for our purposes — a vertical
        // edge reads the same as a horizontal edge with sign flipped).
        const energy = new Float32Array(n);
        const edgeHisto = new Float32Array(n * 8);
        const edgeDir = new Uint8Array(n);
        for (let r = 0; r < rows; r++) {
            for (let c = 0; c < cols; c++) {
                const k = r * cols + c;
                const l = lum[k];
                const lu = lum[Math.max(0, r - 1) * cols + c];
                const ld = lum[Math.min(rows - 1, r + 1) * cols + c];
                const ll = lum[r * cols + Math.max(0, c - 1)];
                const lr = lum[r * cols + Math.min(cols - 1, c + 1)];
                const gx = (lr - ll) * 0.5;
                const gy = (ld - lu) * 0.5;
                const mag = Math.sqrt(gx * gx + gy * gy);
                energy[k] = clamp01(mag / 0.16);
                if (mag > 1e-4) {
                    let theta = (Math.atan2(gy, gx) * 180) / Math.PI;
                    if (theta < 0) theta += 180; // π-periodic
                    const bucket = clamp(Math.floor((theta / 180) * 8), 0, 7);
                    edgeHisto[k * 8 + bucket] += 1;
                    edgeDir[k] = bucket;
                }
            }
        }

        const calm = new Float32Array(n);
        for (let k = 0; k < n; k++) calm[k] = clamp01(1 - energy[k]);

        // ── Pass 3 · subject bounding box (cells significantly busier) ─────
        const eMean = meanOf(energy);
        const eStd = stdOf(energy, eMean);
        const thr = eMean + 0.55 * eStd;
        let x0 = 1;
        let y0 = 1;
        let x1 = 0;
        let y1 = 0;
        let hits = 0;
        const silhouette = new Uint8Array(n);
        for (let r = 0; r < rows; r++) {
            for (let c = 0; c < cols; c++) {
                const k = r * cols + c;
                if (energy[k] > thr) {
                    silhouette[k] = 1;
                    hits++;
                    x0 = Math.min(x0, c / cols);
                    y0 = Math.min(y0, r / rows);
                    x1 = Math.max(x1, (c + 1) / cols);
                    y1 = Math.max(y1, (r + 1) / rows);
                }
            }
        }
        const subjectBox = hits >= 5 ? { x0, y0, x1, y1 } : null;

        // ── Pass 4 · head region (top of subject + warm/skin hue cluster) ──
        // Best-effort. The head is the topmost contiguous run of subject
        // cells whose hue falls in the warm-skin range (10..50°). When no
        // warm cluster exists (B&W, duotone, illustration), we fall back
        // to the top 30% of the subject box.
        let headBox: { x0: number; y0: number; x1: number; y1: number } | null = null;
        if (subjectBox) {
            let hx0 = 1;
            let hy0 = 1;
            let hx1 = 0;
            let hy1 = 0;
            let hHits = 0;
            for (let r = 0; r < rows; r++) {
                for (let c = 0; c < cols; c++) {
                    const k = r * cols + c;
                    if (!silhouette[k]) continue;
                    const { r: rr, g: gg, b: bb } = cellRGB[k];
                    const { h, s } = rgbToHsv(rr, gg, bb);
                    if (
                        r / rows < subjectBox.y0 + (subjectBox.y1 - subjectBox.y0) * 0.4 &&
                        s > 0.12 &&
                        h >= 5 &&
                        h <= 55
                    ) {
                        hHits++;
                        hx0 = Math.min(hx0, c / cols);
                        hy0 = Math.min(hy0, r / rows);
                        hx1 = Math.max(hx1, (c + 1) / cols);
                        hy1 = Math.max(hy1, (r + 1) / rows);
                    }
                }
            }
            if (hHits >= 3) {
                headBox = { x0: hx0, y0: hy0, x1: hx1, y1: hy1 };
            } else if (subjectBox) {
                const h0 = subjectBox.y0;
                const h1 = subjectBox.y0 + (subjectBox.y1 - subjectBox.y0) * 0.3;
                headBox = { x0: subjectBox.x0, y0: h0, x1: subjectBox.x1, y1: h1 };
            }
        }

        // ── Pass 5 · distance-from-subject flood (Chebyshev BFS) ───────────
        // Used to enforce clear-space around the subject: any text placed
        // within `d` cells of a subject cell is at risk of colliding with
        // the silhouette. The flood is single-pass because we iterate to
        // a fixed distance (subject is small relative to the canvas).
        const subjectDistance = new Float32Array(n).fill(64);
        const queue: number[] = [];
        for (let k = 0; k < n; k++) {
            if (silhouette[k]) {
                subjectDistance[k] = 0;
                queue.push(k);
            }
        }
        let head = 0;
        while (head < queue.length) {
            const k = queue[head++];
            const r = (k / cols) | 0;
            const c = k % cols;
            const nd = subjectDistance[k] + 1;
            for (let dr = -1; dr <= 1; dr++) {
                for (let dc = -1; dc <= 1; dc++) {
                    if (!dr && !dc) continue;
                    const nr = r + dr;
                    const nc = c + dc;
                    if (nr < 0 || nc < 0 || nr >= rows || nc >= cols) continue;
                    const nk = nr * cols + nc;
                    if (subjectDistance[nk] > nd) {
                        subjectDistance[nk] = nd;
                        if (nd < 8) queue.push(nk);
                    }
                }
            }
        }

        // ── Pass 6 · dominant colour palette (k=5) via histogram peaks ────
        // We quantize each cell's mean colour to a 32³ RGB cube and tally a
        // histogram. The five highest peaks (after merging near-duplicates)
        // form the palette. This is faster than k-means and equally usable
        // for layout decisions.
        const palHist = new Map<number, number>();
        for (let k = 0; k < n; k++) {
            const { r, g, b } = cellRGB[k];
            const qr = quantize(r, 5);
            const qg = quantize(g, 5);
            const qb = quantize(b, 5);
            const key = (qr << 10) | (qg << 5) | qb;
            palHist.set(key, (palHist.get(key) ?? 0) + 1);
        }
        const sortedPeaks = [...palHist.entries()]
            .sort((a, b) => b[1] - a[1])
            .slice(0, 12)
            .map(([key]) => {
                const qr = (key >> 10) & 31;
                const qg = (key >> 5) & 31;
                const qb = key & 31;
                return rgbToHex(
                    (qr / 4) * 255,
                    (qg / 4) * 255,
                    (qb / 4) * 255,
                );
            });
        const palette: string[] = [];
        for (const c of sortedPeaks) {
            const { r: r1, g: g1, b: b1 } = hexToRgb(c);
            let dup = false;
            for (const p of palette) {
                const { r: r2, g: g2, b: b2 } = hexToRgb(p);
                if (
                    Math.abs(r1 - r2) < 0.08 &&
                    Math.abs(g1 - g2) < 0.08 &&
                    Math.abs(b1 - b2) < 0.08
                ) {
                    dup = true;
                    break;
                }
            }
            if (!dup) {
                palette.push(c);
                if (palette.length === 5) break;
            }
        }

        // ── Pass 7 · column / row marginal calmness profiles ──────────────
        // Interior-only (we skip 2 cells at the top/bottom or left/right to
        // avoid edge artefacts polluting the signal).
        const colProfile = new Float32Array(cols);
        const rowProfile = new Float32Array(rows);
        for (let c = 0; c < cols; c++) {
            let s = 0;
            for (let r = 2; r < rows - 2; r++) s += calm[r * cols + c];
            colProfile[c] = s / (rows - 4);
        }
        for (let r = 0; r < rows; r++) {
            let s = 0;
            for (let c = 2; c < cols - 2; c++) s += calm[r * cols + c];
            rowProfile[r] = s / (cols - 4);
        }

        // ── Pass 8 · vertical / horizontal rhythm signatures ───────────────
        // We compute the autocorrelation of the row and column profiles at
        // lags 1..(n/3). Peaks at regular intervals indicate a periodic
        // rhythm (Swiss grid) — used by the 'fragments' style detector.
        const vRhythm = new Float32Array(Math.floor(rows / 3));
        for (let lag = 1; lag <= vRhythm.length; lag++) {
            let s = 0;
            for (let r = 0; r + lag < rows; r++) s += rowProfile[r] * rowProfile[r + lag];
            vRhythm[lag - 1] = s / (rows - lag);
        }
        const hRhythm = new Float32Array(Math.floor(cols / 3));
        for (let lag = 1; lag <= hRhythm.length; lag++) {
            let s = 0;
            for (let c = 0; c + lag < cols; c++) s += colProfile[c] * colProfile[c + lag];
            hRhythm[lag - 1] = s / (cols - lag);
        }

        // ── Pass 9 · vignette factor (corner vs centre luminance gap) ──────
        let cornerLum = 0;
        let cornerN = 0;
        let centerLum = 0;
        let centerN = 0;
        for (let r = 0; r < rows; r++) {
            for (let c = 0; c < cols; c++) {
                const k = r * cols + c;
                const isCorner =
                    (r < 3 || r >= rows - 3) && (c < 3 || c >= cols - 3);
                const isCenter =
                    r >= rows * 0.35 &&
                    r < rows * 0.65 &&
                    c >= cols * 0.35 &&
                    c < cols * 0.65;
                if (isCorner) {
                    cornerLum += lum[k];
                    cornerN++;
                }
                if (isCenter) {
                    centerLum += lum[k];
                    centerN++;
                }
            }
        }
        const vignette = clamp01(
            Math.abs(cornerLum / Math.max(1, cornerN) - centerLum / Math.max(1, centerN)) * 2,
        );

        // ── Pass 10 · text-side bias (which half is calmer) ────────────────
        let leftCalm = 0;
        let rightCalm = 0;
        for (let c = 0; c < 6; c++) leftCalm += colProfile[c];
        for (let c = cols - 6; c < cols; c++) rightCalm += colProfile[c];
        const textSide: 'left' | 'right' = leftCalm >= rightCalm ? 'left' : 'right';

        // ── Pass 11 · high-contrast monochrome detection ─────────────────
        // The 'noir' style is appropriate when saturation is very low AND
        // the luminance histogram has heavy tails (deep blacks + bright
        // highlights). We approximate the latter with the std-dev of the
        // luminance field.
        const satAvg = satSum / Math.max(1, satCnt);
        const lumStd = stdOf(lum);
        const highContrastMono = satAvg < 0.1 && lumStd > 0.28;

        let hue = Math.atan2(sinH, cosH) * (180 / Math.PI);
        if (hue < 0) hue += 360;

        return {
            cols,
            rows,
            lum,
            energy,
            sat,
            calm,
            edgeHisto,
            edgeDir,
            silhouette,
            subjectDistance,
            light: lumSum / Math.max(1, n) > 0.58,
            subjectBox,
            headBox,
            hue,
            satAvg,
            palette,
            colProfile,
            rowProfile,
            vRhythm,
            hRhythm,
            textSide,
            vignette,
            highContrastMono,
        };
    } catch {
        return null;
    }
}

/**
 * Project the field into the composer's PhotoAnalysis shape. ONE scan feeds
 * both engines, so ASSEMBLE's opening step is a single 96×120 pass.
 */
export function gridToPhotoAnalysis(g: GridAnalysis): PhotoAnalysis {
    const bandCalm = (
        x0: number,
        y0: number,
        x1: number,
        y1: number,
    ): number => {
        let s = 0;
        let n = 0;
        for (let r = 0; r < g.rows; r++) {
            for (let c = 0; c < g.cols; c++) {
                const cc = (c + 0.5) / g.cols;
                const rr = (r + 0.5) / g.rows;
                if (cc < x0 || cc > x1 || rr < y0 || rr > y1) continue;
                s += g.calm[r * g.cols + c];
                n++;
            }
        }
        return Math.round((s / Math.max(1, n)) * 100) / 100;
    };
    const cx = g.subjectBox
        ? (g.subjectBox.x0 + g.subjectBox.x1) / 2
        : 0.5;
    return {
        headroom: bandCalm(0, 0, 1, 0.18),
        leftSpace: bandCalm(0, 0.18, 0.24, 0.72),
        rightSpace: bandCalm(0.76, 0.18, 1, 0.72),
        bottomSpace: bandCalm(0, 0.74, 1, 0.94),
        photoLight: g.light,
        subjectX: cx < 0.42 ? 'left' : cx > 0.58 ? 'right' : 'center',
        hue: Math.round(g.hue),
        sat: Math.round(g.satAvg * 100) / 100,
    };
}

const zoneMetrics = (
    g: GridAnalysis,
    id: string,
    x0: number,
    y0: number,
    x1: number,
    y1: number,
    tags: string[],
): Zone => {
    let calmSum = 0;
    let lumSum = 0;
    let n = 0;
    for (let r = 0; r < g.rows; r++) {
        for (let c = 0; c < g.cols; c++) {
            const cc = (c + 0.5) / g.cols;
            const rr = (r + 0.5) / g.rows;
            if (cc < x0 || cc > x1 || rr < y0 || rr > y1) continue;
            const k = r * g.cols + c;
            calmSum += g.calm[k];
            lumSum += g.lum[k];
            n++;
        }
    }
    n = Math.max(1, n);
    const lumMean = lumSum / n;
    let varAcc = 0;
    for (let r = 0; r < g.rows; r++) {
        for (let c = 0; c < g.cols; c++) {
            const cc = (c + 0.5) / g.cols;
            const rr = (r + 0.5) / g.rows;
            if (cc < x0 || cc > x1 || rr < y0 || rr > y1) continue;
            const dl = g.lum[r * g.cols + c] - lumMean;
            varAcc += dl * dl;
        }
    }
    const w = x1 - x0;
    const h = y1 - y0;
    return {
        id,
        x0,
        y0,
        x1,
        y1,
        w,
        h,
        cx: (x0 + x1) / 2,
        cy: (y0 + y1) / 2,
        calm: calmSum / n,
        flat: clamp01(1 - Math.sqrt(varAcc / n) * 3),
        lum: lumMean,
        aspect: w / Math.max(0.02, h),
        tags,
    };
};

/** Returns true when zone `z` is structurally allowed for component `compId`
 *  given the current style — i.e. style-specific hard constraints pass. */
export const zoneAllowedFor = (
    z: Zone,
    compId: string,
    style: SystemStyleId,
): boolean => {
    // Universal hard constraints.
    if (compId === 'badge' && !z.tags.includes('corner')) return false;
    if (compId === 'scan' && !z.tags.includes('corner')) return false;
    if (compId === 'crosshair' && !z.tags.includes('corner') && !z.tags.includes('edge')) return false;
    if (compId === 'divider' && !z.tags.includes('edge')) return false;
    if (compId === 'rail' && !z.tags.includes('rail') && !z.tags.includes('vertical-rail')) return false;
    if (compId === 'giantType' && !z.tags.includes('giant')) return false;

    // Style-specific constraints.
    switch (style) {
        case 'fragments':
            // HUMAN FRAGMENTS: masthead lives in giant type, cover lines in
            // the vertical rail, dateline in the bottom of the rail.
            if (compId === 'masthead' && !z.tags.includes('giant')) return false;
            if (compId === 'feature' && !z.tags.includes('vertical-rail')) return false;
            if (compId === 'sub' && !z.tags.includes('vertical-rail')) return false;
            if (compId === 'teasers' && !z.tags.includes('vertical-rail')) return false;
            if (compId === 'dateline' && !z.tags.includes('vertical-rail')) return false;
            return true;
        case 'vogue':
            if (compId === 'masthead' && !z.tags.some((t) => ['top', 'band'].includes(t))) return false;
            if (compId === 'feature' && z.cy > 0.7) return false;
            return true;
        case 'time':
            if (compId === 'masthead' && z.cy > 0.2) return false;
            return true;
        case 'natgeo':
            // National Geographic: masthead top-left, minimal text, full border.
            if (compId === 'masthead' && !z.tags.includes('top')) return false;
            if (compId === 'feature' && z.cy < 0.8) return false;
            return true;
        case 'noir':
            if (compId === 'masthead' && !z.tags.includes('top')) return false;
            return true;
        case 'kinfolk':
            // Kinfolk: minimal, lots of whitespace.
            if (compId === 'masthead' && z.cy > 0.3) return false;
            if (compId === 'feature' && z.cy < 0.5) return false;
            return true;
        default:
            return true;
    }
};

/** Auto-select the best-fit style for a given field measurement. Returns
 *  the style with the highest confidence match, breaking ties by registry
 *  order. */
export const autoSelectStyle = (g: GridAnalysis): SystemStyleId => {
    let best: SystemStyleId = 'fragments' as SystemStyleId;
    let bestScore = -1;
    for (const p of STYLE_PRESETS) {
        const s = p.matches(g);
        if (s > bestScore) {
            bestScore = s;
            best = p.id;
        }
    }
    return best;
};

// ===========================================================================
// SECTION 4 · THE ASSIGNMENT OPTIMIZER (style-aware Hungarian)
// ---------------------------------------------------------------------------
// Classic Jonker-style Hungarian assignment with potentials. Cost matrix is
// style-aware: the same component × zone pair can be cheap in one style
// and infeasible in another. Hard constraints come from `zoneAllowedFor`
// (universal) and the style's `forbiddenZones` map; soft preferences come
// from the style's `preferredZones` map and `weights` vector.
// ===========================================================================

const INFEASIBLE = 1e6;

/** Jonker-style Hungarian assignment with potentials. cost[r][c] is an
 *  n×n matrix (we pad with phantom rows/cols as needed). Returns one col
 *  index per row, minimizing total cost. Classic, exact, unbeatable. */
const hungarian = (cost: number[][]): number[] => {
    const n = cost.length;
    if (n === 0) return [];
    const INF = 1e15;
    const u = new Array<number>(n + 1).fill(0);
    const v = new Array<number>(n + 1).fill(0);
    const p = new Array<number>(n + 1).fill(0);
    const way = new Array<number>(n + 1).fill(0);
    for (let i = 1; i <= n; i++) {
        p[0] = i;
        let j0 = 0;
        const minv = new Array<number>(n + 1).fill(INF);
        const used = new Array<boolean>(n + 1).fill(false);
        do {
            used[j0] = true;
            const i0 = p[j0];
            let delta = INF;
            let j1 = 0;
            for (let j = 1; j <= n; j++) {
                if (used[j]) continue;
                const cur = cost[i0 - 1][j - 1] - u[i0] - v[j];
                if (cur < minv[j]) {
                    minv[j] = cur;
                    way[j] = j0;
                }
                if (minv[j] < delta) {
                    delta = minv[j];
                    j1 = j;
                }
            }
            for (let j = 0; j <= n; j++) {
                if (used[j]) {
                    u[p[j]] += delta;
                    v[j] -= delta;
                } else {
                    minv[j] -= delta;
                }
            }
            j0 = j1;
        } while (p[j0] !== 0);
        do {
            const j1 = way[j0];
            p[j0] = p[j1];
            j0 = j1;
        } while (j0);
    }
    const rowToCol = new Array<number>(n).fill(-1);
    for (let j = 1; j <= n; j++) if (p[j] > 0) rowToCol[p[j] - 1] = j - 1;
    return rowToCol;
};

/** Compute the contrast ratio of a candidate text colour against the
 *  average luminance of a zone. Returns ∞ when the zone is perfectly flat
 *  against the text colour. */
const zoneContrastFor = (z: Zone, textHex: string): number => {
    const { r, g, b } = hexToRgb(textHex);
    const lText = luminanceOf(r, g, b);
    return contrastRatio(lText, z.lum);
};

/** Compute the cost of assigning `compId` to zone `z` under style `preset`.
 *  The cost is a weighted sum of:
 *    • zone calmness (busy zones cost more)
 *    • aspect mismatch (log of aspect ratio gap)
 *    • zone flatness (noisy zones cost more)
 *    • subject adjacency (overlap with subject box — sign depends on style)
 *    • grid alignment (distance from 24-division)
 *    • contrast against zone luminance
 *    • role prior (preferred zones are cheaper, forbidden are infeasible)
 *    • flatness bonus (very flat zones get a discount)
 *  Plus style-specific tweaks (masthead prefer top, feature prefer near
 *  subject, badge/scan must be in corner, etc.). */
const costOf = (
    comp: ComponentSpec,
    z: Zone,
    g: GridAnalysis,
    preset: StylePreset,
    jitter: () => number,
): number => {
    if (z.calm < comp.minCalm) return INFEASIBLE;
    if (!zoneAllowedFor(z, comp.id, preset.id)) return INFEASIBLE;
    const forbidden = preset.forbiddenZones[comp.id] ?? [];
    if (forbidden.includes(z.id)) return INFEASIBLE;

    const w = preset.weights;
    const zoneAspect = z.w / Math.max(0.02, z.h);

    let c =
        w.calm * (1 - z.calm) * jitter() +
        w.aspect * Math.abs(Math.log(zoneAspect / comp.aspect)) * jitter() +
        w.flat * (1 - z.flat) +
        w.gridAlignment * Math.abs(snap24(z.cx) - z.cx) * 4 +
        w.gridAlignment * Math.abs(snap24(z.cy) - z.cy) * 4;

    // Subject adjacency — sign depends on whether this component wants to
    // touch the subject (cover lines point AT subject; masthead may sit
    // behind; barcode must NOT touch).
    if (g.subjectBox) {
        const overlap = overlapRatio(z, g.subjectBox);
        switch (comp.id) {
            case 'masthead':
                if (preset.masthead.allowBehindSubject) {
                    c += w.subjectAdjacency * overlap * 0.6;
                } else {
                    c += w.subjectAdjacency * overlap * 2.0;
                }
                break;
            case 'feature':
            case 'sub':
                // Cover lines prefer to be NEAR the subject but not ON it.
                c += w.subjectAdjacency * Math.abs(overlap - 0.15);
                break;
            case 'teasers':
                c += w.subjectAdjacency * overlap * 2.0;
                break;
            case 'scan':
            case 'badge':
                c += w.subjectAdjacency * overlap * 4.0;
                break;
        }
    }

    // Contrast — penalize text colours that don't meet the style's minimum.
    const textHex = preset.colours.bodyColor;
    const contrast = zoneContrastFor(z, textHex);
    const minContrast = preset.colours.minContrast;
    if (contrast < minContrast) {
        c += w.contrast * (minContrast - contrast) * 2;
    } else {
        c -= w.flatnessBonus * (z.flat > 0.8 ? 0.4 : 0);
    }

    // Role priors — style-specific preferred zones are cheaper.
    const preferred = preset.preferredZones[comp.id] ?? [];
    const prefIdx = preferred.indexOf(z.id);
    if (prefIdx >= 0) {
        c -= w.rolePrior * (1 - prefIdx * 0.2);
    }

    // Component-specific tweaks — same shape as v1 but style-aware.
    switch (comp.id) {
        case 'masthead':
            if (preset.masthead.placement === 'giant-as-canvas') {
                // For HUMAN FRAGMENTS, masthead IS the giant type.
                if (!z.tags.includes('giant')) return INFEASIBLE;
                return 0.1; // already filtered to giant zone
            }
            if (preset.masthead.placement === 'top-full-over' ||
                preset.masthead.placement === 'top-full-behind') {
                if (z.cy > 0.25) c += 1.2;
            }
            if (preset.masthead.placement === 'top-center' && z.cx < 0.3) c += 0.6;
            if (preset.masthead.placement === 'top-left' && z.cx > 0.6) c += 0.6;
            if (preset.masthead.placement === 'bottom-center' && z.cy < 0.7) c += 1.5;
            break;
        case 'dateline':
            // Dateline must hug a corner — TL, TR, BL, or BR.
            if (!z.tags.includes('dateline') && !z.tags.includes('corner')) c += 0.6;
            if (preset.coverLines.columnStrategy === 'vertical-rail-right' &&
                !z.tags.includes('vertical-rail')) return INFEASIBLE;
            break;
        case 'feature':
        case 'sub':
            // For vertical-rail styles, both must live in the rail.
            if (preset.coverLines.columnStrategy === 'vertical-rail-right') {
                if (!z.tags.includes('vertical-rail')) return INFEASIBLE;
                return 0.2;
            }
            if (preset.coverLines.columnStrategy === 'vertical-rail-left') {
                if (!z.tags.includes('vertical-rail')) return INFEASIBLE;
                return 0.2;
            }
            // For 'below-chin' main line position, prefer BC zone.
            if (preset.coverLines.mainLinePosition === 'below-chin' &&
                z.id === 'BC') c -= 0.8;
            if (preset.coverLines.mainLinePosition === 'at-bottom' &&
                (z.id === 'B1' || z.id === 'B3')) c -= 0.8;
            if (preset.coverLines.mainLinePosition === 'to-the-side' &&
                (z.id === 'SL' || z.id === 'SR')) c -= 0.8;
            break;
        case 'teasers':
            if (preset.coverLines.columnStrategy === 'vertical-rail-right') {
                if (!z.tags.includes('vertical-rail')) return INFEASIBLE;
                return 0.3;
            }
            if (z.h < 0.2) c += 0.6;
            break;
        case 'badge':
            if (!z.tags.includes('corner')) return INFEASIBLE;
            break;
        case 'scan':
            if (!z.tags.includes('corner')) return INFEASIBLE;
            // Scan goes to the corner opposite the text mass.
            if (g.textSide === 'left' && z.tags.includes('left')) c += 0.8;
            if (g.textSide === 'right' && z.tags.includes('right')) c += 0.8;
            break;
        case 'giantType':
            if (!z.tags.includes('giant')) return INFEASIBLE;
            if (!preset.usesGiantType) return INFEASIBLE;
            // Prefer the full-canvas giant zone.
            if (z.id === 'GF') return 0.1;
            break;
        case 'rail':
            if (!preset.usesVerticalRail) return INFEASIBLE;
            if (!z.tags.includes('rail') && !z.tags.includes('vertical-rail')) return INFEASIBLE;
            // Prefer right rail by default.
            if (z.tags.includes('right')) c -= 0.4;
            break;
        case 'crosshair':
            if (!preset.usesCrosshairs) return INFEASIBLE;
            if (!z.tags.includes('corner') && !z.tags.includes('edge')) return INFEASIBLE;
            // Place crosshairs in diagonal corners (TL + BR by default).
            if (z.id === 'E1' || z.id === 'E4') c -= 1.2;
            if (z.id === 'E2' || z.id === 'E3') c += 0.8;
            break;
        case 'accentBlock':
            if (preset.decorations.set !== 'accent-block') return INFEASIBLE;
            // Place accent block in lower-mid area, behind giant type.
            if (z.id === 'G3' || z.id === 'G2') c -= 0.5;
            break;
        case 'divider':
            if (!preset.usesEmDashDividers) return INFEASIBLE;
            if (!z.tags.includes('edge')) return INFEASIBLE;
            // Dividers live on the rail edges (left or right edge mid).
            if (z.id === 'X2' || z.id === 'X4') c -= 0.4;
            break;
    }

    return c;
};

// SECTION 5 · THE GEOMETRY ENGINE
// ---------------------------------------------------------------------------
// Resolves a solved assignment into actual placement coordinates snapped
// to the optical grid. Handles five geometry sub-systems:

//   5.1  Standard placement      — masthead, dateline, cover lines

//   5.5  Vertical-rail stack     — HUMAN FRAGMENTS right-rail info stack
//   5.6  Crosshair decoration    — HUMAN FRAGMENTS + registration marks
//   5.7  Accent colour block     — HUMAN FRAGMENTS red block accent
//   5.8  Em-dash dividers        — HUMAN FRAGMENTS — separators
//   5.9  Border frame            — TIME / NatGeo / Cyber frames

// --- 5.1 · Standard placement ----------------------------------------------

/** Resolve a component's solved zone into a placement rectangle snapped
 *  to the 24-division optical grid. Handles vertical-rail rotation and
 *  plate necessity. */
const resolvePlacement = (
    comp: ComponentSpec,
    z: Zone,
    g: GridAnalysis,
    preset: StylePreset,
): SolvedPlacement => {
    // Vertical rail masthead — text rotates 90° to fit a tall narrow zone.
    const isRail =
        comp.id === 'masthead' &&
        preset.masthead.allowVertical &&
        z.tags.includes('rail') &&
        z.h / Math.max(0.04, z.w) >= 1.9;

    let widthPct: number;
    let heightPct: number;
    let rotationDeg = 0;

    if (isRail) {
        widthPct = clamp(z.h * 100 * 0.85, 46, 64);
        heightPct = clamp(z.w * 100 * 0.6, 4, 8);
        rotationDeg = z.tags.includes('left') ? -90 : 90;
    } else if (comp.id === 'masthead' && preset.usesGiantType) {
        // Giant type fills the canvas — full height, full width minus rail.
        widthPct = preset.usesVerticalRail ? 75 : 92;
        heightPct = preset.typeScale.masthead * 100;
        rotationDeg = 0;
    } else {
        // SCALE BELONGS TO THE ROLE — the zone's width is a ceiling, but
        // the role sets the floor: a masthead never renders as a slim
        // afterthought even when its calm zone is narrow, and heights
        // come from the preset's typeScale ladder (the editorial
        // hierarchy contract), not from the zone's pocket size.
        const sizeScale =
            comp.id === 'masthead' ? 0.92 :
            comp.id === 'feature' ? 0.88 :
            comp.id === 'sub' ? 0.85 :
            0.82;
        const minW =
            comp.id === 'masthead' ? 46 :
            comp.id === 'feature' ? 34 :
            comp.id === 'sub' ? 24 :
            18;
        const maxW =
            comp.id === 'masthead' ? 92 :
            comp.id === 'feature' ? 78 :
            60;
        widthPct = clamp(z.w * 100 * sizeScale, minW, maxW);
        const ts = preset.typeScale;
        const roleScale =
            comp.id === 'masthead' ? ts.masthead :
            comp.id === 'feature' ? ts.feature :
            comp.id === 'sub' ? ts.sub :
            comp.id === 'teasers' ? ts.teasers :
            comp.id === 'dateline' ? ts.dateline :
            ts.teasers;
        heightPct = clamp(roleScale * 100, 2.4, 46);
    }

    const cx = isRail
        ? clamp(z.x0 + (z.tags.includes('left') ? 0.045 : -0.045) + z.w, 0.03, 0.97)
        : opticalSnap(clamp(z.cx, 0.06, 0.94));
    const cy = opticalSnap(clamp(z.cy, 0.08, 0.92));

    const plate =
        (comp.id === 'feature' || comp.id === 'sub') &&
        !!g.subjectBox &&
        overlapRatio(z, g.subjectBox) > 0.35 &&
        z.calm < 0.52;

    const depth: 'behind' | 'front' =
        comp.id === 'masthead' &&
        g.subjectBox &&
        overlapRatio(z, g.subjectBox) > 0.4 &&
        preset.masthead.allowBehindSubject
            ? 'behind'
            : 'front';

    return {
        role: comp.id,
        zone: z.id,
        xPct: Math.round(cx * 1000) / 1000,
        yPct: Math.round(cy * 1000) / 1000,
        widthPct: Math.round(widthPct),
        heightPct: Math.round(heightPct * 10) / 10,
        rotationDeg,
        plate,
        depth,
        maskPhoto: preset.colours.treatment === 'mask-in-type' && comp.id === 'masthead',
    };
};

// --- 5.2 · Plate geometry ---------------------------------------------------

/** Add a solid backdrop plate behind a text spec when the zone is too busy
 *  for legible text without one. Plate colour inverts against photo mean
 *  luminance so the text remains readable. */
const buildPlate = (
    feat: SolvedPlacement,
    photoLight: boolean,
): DecorationShape => ({
    kind: 'rail-bg',
    xPct: feat.xPct - 0.04,
    yPct: feat.yPct - 0.015,
    wPct: feat.widthPct + 8,
    hPct: feat.heightPct + 2.5,
    color: photoLight ? '#000000' : '#FFFFFF',
    rotationDeg: 0,
    depth: 'behind',
});

// --- 5.4 · Giant-type-as-mask geometry -------------------------------------

/** For the HUMAN FRAGMENTS style: render the masthead word as giant
 *  letterforms stacked vertically, with the photo either clipped inside
 *  the letterforms (mask-in-type) or overlaying them (overlay-on-type).
 *  The word is sliced into N bands, one per letter, each occupying an
 *  equal vertical slice of the canvas. */
const buildGiantType = (
    preset: StylePreset,
    word: string,
): DecorationShape => {
    const letters = (word || 'PULSE').toUpperCase().split('').slice(0, 6);
    return {
        kind: 'giant-type',
        xPct: 0.04,
        yPct: 0.05,
        wPct: preset.usesVerticalRail ? 0.74 : 0.92,
        hPct: 0.9,
        color: preset.colours.mastheadColor ?? '#1A1A1A',
        rotationDeg: 0,
        depth: 'behind',
        text: letters.join(''),
        typeFamily: preset.masthead.typeFamily,
        weight: preset.masthead.weight,
        letterSpacing: preset.masthead.letterSpacing,
        maskPhoto: preset.colours.treatment === 'mask-in-type',
    };
};

// --- 5.5 · Vertical-rail stack ----------------------------------------------

/** The four-block vertical info rail: title → series → tagline → year,
 *  separated by em-dash dividers. Each block snaps to the rail's baseline
 *  grid (the 24-division optical grid). */
const buildVerticalRail = (
    preset: StylePreset,
    semantics: SolverSemantics | null,
): { placements: SolvedPlacement[]; decorations: DecorationShape[] } => {
    const railX = 0.875; // centre of right rail
    const railW = 18; // percent width
    const blocks = [
        { role: 'feature', label: semantics?.subjectFocus ?? 'PULSE FORM', size: 0.038 },
        { role: 'sub', label: 'HUMAN FRAGMENTS', size: 0.022 },
        { role: 'teasers', label: semantics?.mood?.[0] ?? 'presence shapes space', size: 0.018 },
        { role: 'dateline', label: '2026', size: 0.018 },
    ];
    const placements: SolvedPlacement[] = [];
    const decorations: DecorationShape[] = [];
    // Distribute blocks vertically: top of rail at 12%, bottom at 88%.
    const top = 0.12;
    const bottom = 0.88;
    const step = (bottom - top) / (blocks.length + (preset.usesEmDashDividers ? blocks.length - 1 : 0) + 1);
    let y = top + step;
    blocks.forEach((b, i) => {
        placements.push({
            role: b.role,
            zone: 'VR',
            xPct: railX,
            yPct: Math.round(y * 1000) / 1000,
            widthPct: railW,
            heightPct: Math.round(b.size * 100 * 10) / 10,
            rotationDeg: 0,
            plate: false,
            depth: 'front',
            maskPhoto: false,
            railSlot: i,
        });
        y += step;
        // Em-dash divider between blocks.
        if (preset.usesEmDashDividers && i < blocks.length - 1) {
            decorations.push({
                kind: 'em-dash',
                xPct: railX,
                yPct: Math.round(y * 1000) / 1000,
                wPct: railW * 0.4,
                hPct: 0.4,
                color: preset.colours.bodyColor,
                rotationDeg: 0,
                depth: 'front',
                thickness: 0.18,
            });
            y += step;
        }
    });
    return { placements, decorations };
};

// --- 5.6 · Crosshair decoration -------------------------------------------

/** Place the HUMAN FRAGMENTS + registration marks in diagonal corners.
 *  Default: TL + BR. The size and arm length follow the style preset. */
const buildCrosshairs = (preset: StylePreset): DecorationShape[] => {
    if (!preset.usesCrosshairs) return [];
    const size = preset.decorations.crosshairSizePct;
    const arm = size * 0.6;
    return preset.decorations.crosshairSlots.map((slot) => {
        const pos: Record<string, { x: number; y: number }> = {
            TL: { x: 0.08, y: 0.08 },
            TR: { x: 0.92, y: 0.08 },
            BL: { x: 0.08, y: 0.92 },
            BR: { x: 0.92, y: 0.92 },
        };
        const p = pos[slot];
        return {
            kind: 'crosshair',
            xPct: p.x,
            yPct: p.y,
            wPct: size * 100,
            hPct: size * 100,
            color: preset.colours.bodyColor,
            rotationDeg: 0,
            depth: 'front',
            arm,
        };
    });
};

// --- 5.7 · Accent colour block ---------------------------------------------

/** The HUMAN FRAGMENTS accent colour block — a single rectangle of the
 *  accent colour placed behind the giant type, mid-canvas. Optional and
 *  only rendered when the style's decoration set is 'accent-block'. */
const buildAccentBlock = (preset: StylePreset): DecorationShape | null => {
    if (!preset.colours.accent || !preset.decorations.accentBlockPos) return null;
    return {
        kind: 'accent-block',
        xPct: preset.decorations.accentBlockPos.x,
        yPct: preset.decorations.accentBlockPos.y,
        wPct: preset.decorations.accentBlockW * 100,
        hPct: preset.decorations.accentBlockH * 100,
        color: preset.colours.accent,
        rotationDeg: 0,
        depth: 'behind',
    };
};

// --- 5.9 · Border frame -----------------------------------------------------

/** The TIME / NatGeo / Cyber coloured border frame. Renders as a hollow
 *  rectangle around the cover, with thickness from the style preset. */
const buildBorderFrame = (preset: StylePreset): DecorationShape | null => {
    if (!preset.decorations.frameColor) return null;
    const t = preset.decorations.frameThicknessPct;
    return {
        kind: 'frame',
        xPct: t / 2,
        yPct: t / 2,
        wPct: 100 - t,
        hPct: 100 - t,
        color: preset.decorations.frameColor,
        rotationDeg: 0,
        depth: 'front',
        thickness: t * 100,
        corner: 0,
    };
};

// ===========================================================================
// SECTION 6 · THE LAYOUT-LAW ENGINE (fifteen physical rules)
// ---------------------------------------------------------------------------
// After the Hungarian solver returns a placement set, the layout-law engine
// verifies that every rule is satisfied. Violations are returned — they do
// NOT silently mutate the placement. The composer may then either accept
// the violations (with a warning badge) or request a re-solve with a new
// seed.
// ===========================================================================

export interface LayoutViolation {
    ruleId: string;
    severity: 'critical' | 'warning' | 'info';
    message: string;
    affectedRole?: string;
    affectedZone?: string;
}

/** The fifteen layout laws. Each takes the placements, the field, and the
 *  preset, and returns zero or more violations. */
const LAYOUT_LAWS: Array<{
    id: string;
    severity: 'critical' | 'warning' | 'info';
    check: (
        placements: SolvedPlacement[],
        g: GridAnalysis,
        preset: StylePreset,
    ) => LayoutViolation[];
}> = [
    // Law 1 · Minimum clear space around text
    {
        id: 'clear-space',
        severity: 'warning',
        check: (placements, g) => {
            const v: LayoutViolation[] = [];
            for (const p of placements) {
                if (p.role === 'scan' || p.role === 'crosshair' || p.role === 'divider') continue;
                if (!g.subjectBox) continue;
                const textBox = {
                    x0: p.xPct - p.widthPct / 200,
                    y0: p.yPct - p.heightPct / 200,
                    x1: p.xPct + p.widthPct / 200,
                    y1: p.yPct + p.heightPct / 200,
                };
                const ov = overlapRatio(textBox, g.subjectBox);
                if (ov > 0.5 && !p.plate && p.depth === 'front') {
                    v.push({
                        ruleId: 'clear-space',
                        severity: 'warning',
                        message: `${p.role} overlaps subject by ${Math.round(ov * 100)}% without a plate`,
                        affectedRole: p.role,
                        affectedZone: p.zone,
                    });
                }
            }
            return v;
        },
    },
    // Law 2 · Masthead size hierarchy
    {
        id: 'masthead-hierarchy',
        severity: 'critical',
        check: (placements, _g, preset) => {
            const m = placements.find((p) => p.role === 'masthead');
            const f = placements.find((p) => p.role === 'feature');
            if (!m || !f) return [];
            if (preset.usesGiantType) return [];
            if (f.heightPct > m.heightPct * 1.1) {
                return [{
                    ruleId: 'masthead-hierarchy',
                    severity: 'critical',
                    message: `Feature (${f.heightPct}%) larger than masthead (${m.heightPct}%)`,
                    affectedRole: 'feature',
                }];
            }
            return [];
        },
    },
    // Law 3 · Cover lines never overlap masthead
    {
        id: 'cover-line-masthead-overlap',
        severity: 'critical',
        check: (placements) => {
            const v: LayoutViolation[] = [];
            const m = placements.find((p) => p.role === 'masthead');
            if (!m) return v;
            const mbox = {
                x0: m.xPct - m.widthPct / 200,
                y0: m.yPct - m.heightPct / 200,
                x1: m.xPct + m.widthPct / 200,
                y1: m.yPct + m.heightPct / 200,
            };
            for (const p of placements) {
                if (p.role === 'masthead' || p.role === 'scan' ||
                    p.role === 'crosshair' || p.role === 'divider' ||
                    p.role === 'giantType') continue;
                const box = {
                    x0: p.xPct - p.widthPct / 200,
                    y0: p.yPct - p.heightPct / 200,
                    x1: p.xPct + p.widthPct / 200,
                    y1: p.yPct + p.heightPct / 200,
                };
                if (overlapArea(mbox, box) > 0.001) {
                    v.push({
                        ruleId: 'cover-line-masthead-overlap',
                        severity: 'critical',
                        message: `${p.role} overlaps masthead`,
                        affectedRole: p.role,
                    });
                }
            }
            return v;
        },
    },
    // Law 4 · Barcode always in corner
    {
        id: 'barcode-corner',
        severity: 'warning',
        check: (placements) => {
            const s = placements.find((p) => p.role === 'scan');
            if (!s) return [];
            if (s.xPct > 0.18 && s.xPct < 0.82 && s.yPct > 0.18 && s.yPct < 0.82) {
                return [{
                    ruleId: 'barcode-corner',
                    severity: 'warning',
                    message: 'Barcode placed in canvas interior, not corner',
                    affectedRole: 'scan',
                }];
            }
            return [];
        },
    },
    // Law 5 · Dateline adjacent to masthead
    {
        id: 'dateline-adjacency',
        severity: 'info',
        check: (placements) => {
            const m = placements.find((p) => p.role === 'masthead');
            const d = placements.find((p) => p.role === 'dateline');
            if (!m || !d) return [];
            const dy = Math.abs(m.yPct - d.yPct);
            if (dy > 0.18) {
                return [{
                    ruleId: 'dateline-adjacency',
                    severity: 'info',
                    message: 'Dateline far from masthead',
                    affectedRole: 'dateline',
                }];
            }
            return [];
        },
    },
    // Law 6 · Feature never above masthead
    {
        id: 'feature-above-masthead',
        severity: 'critical',
        check: (placements, _g, preset) => {
            if (preset.usesGiantType) return [];
            const m = placements.find((p) => p.role === 'masthead');
            const f = placements.find((p) => p.role === 'feature');
            if (!m || !f) return [];
            if (f.yPct < m.yPct - 0.02) {
                return [{
                    ruleId: 'feature-above-masthead',
                    severity: 'critical',
                    message: 'Feature placed above masthead',
                    affectedRole: 'feature',
                }];
            }
            return [];
        },
    },
    // Law 7 · Vertical text never overlaps horizontal text
    {
        id: 'vertical-text-collision',
        severity: 'warning',
        check: (placements) => {
            const v: LayoutViolation[] = [];
            const vert = placements.filter((p) => Math.abs(p.rotationDeg) === 90);
            const horiz = placements.filter((p) => p.rotationDeg === 0);
            for (const vv of vert) {
                for (const hh of horiz) {
                    const vBox = {
                        x0: vv.xPct - vv.heightPct / 200,
                        y0: vv.yPct - vv.widthPct / 200,
                        x1: vv.xPct + vv.heightPct / 200,
                        y1: vv.yPct + vv.widthPct / 200,
                    };
                    const hBox = {
                        x0: hh.xPct - hh.widthPct / 200,
                        y0: hh.yPct - hh.heightPct / 200,
                        x1: hh.xPct + hh.widthPct / 200,
                        y1: hh.yPct + hh.heightPct / 200,
                    };
                    if (overlapArea(vBox, hBox) > 0.005) {
                        v.push({
                            ruleId: 'vertical-text-collision',
                            severity: 'warning',
                            message: `Vertical ${vv.role} overlaps horizontal ${hh.role}`,
                            affectedRole: vv.role,
                        });
                    }
                }
            }
            return v;
        },
    },
    // Law 8 · Subject visibility — must show ≥ 60% of subject area
    {
        id: 'subject-visibility',
        severity: 'warning',
        check: (placements, g) => {
            if (!g.subjectBox) return [];
            const subjArea = (g.subjectBox.x1 - g.subjectBox.x0) * (g.subjectBox.y1 - g.subjectBox.y0);
            let coveredArea = 0;
            for (const p of placements) {
                if (p.depth !== 'front') continue;
                if (p.role === 'scan' || p.role === 'crosshair' || p.role === 'divider') continue;
                const box = {
                    x0: p.xPct - p.widthPct / 200,
                    y0: p.yPct - p.heightPct / 200,
                    x1: p.xPct + p.widthPct / 200,
                    y1: p.yPct + p.heightPct / 200,
                };
                coveredArea += overlapArea(box, g.subjectBox);
            }
            const ratio = coveredArea / subjArea;
            if (ratio > 0.4) {
                return [{
                    ruleId: 'subject-visibility',
                    severity: 'warning',
                    message: `Subject ${Math.round((1 - ratio) * 100)}% visible (< 60% threshold)`,
                }];
            }
            return [];
        },
    },
    // Law 9 · Colour contrast minimum (WCAG)
    {
        id: 'colour-contrast',
        severity: 'warning',
        check: (placements, g, preset) => {
            const v: LayoutViolation[] = [];
            const textLum = luminanceOf(...Object.values(hexToRgb(preset.colours.bodyColor)) as [number, number, number]);
            for (const p of placements) {
                if (p.role === 'scan' || p.role === 'crosshair' || p.role === 'divider') continue;
                // Sample the zone luminance under the placement.
                const zone = enumerateZones(g).find((z) => z.id === p.zone);
                if (!zone) continue;
                const contrast = contrastRatio(textLum, zone.lum);
                if (contrast < preset.colours.minContrast && !p.plate) {
                    v.push({
                        ruleId: 'colour-contrast',
                        severity: 'warning',
                        message: `${p.role} contrast ${contrast.toFixed(1)} < ${preset.colours.minContrast} (no plate)`,
                        affectedRole: p.role,
                    });
                }
            }
            return v;
        },
    },
    // Law 10 · Optical grid alignment
    {
        id: 'optical-grid',
        severity: 'info',
        check: (placements) => {
            const v: LayoutViolation[] = [];
            for (const p of placements) {
                if (p.role === 'scan' || p.role === 'crosshair') continue;
                if (!nearGrid24(p.xPct) || !nearGrid24(p.yPct)) {
                    v.push({
                        ruleId: 'optical-grid',
                        severity: 'info',
                        message: `${p.role} off 24-division grid`,
                        affectedRole: p.role,
                    });
                }
            }
            return v;
        },
    },
    // Law 11 · Vertical rhythm — rail blocks on baseline grid
    {
        id: 'rail-baseline',
        severity: 'info',
        check: (placements, _g, preset) => {
            if (!preset.usesVerticalRail) return [];
            const rail = placements.filter((p) => p.zone === 'VR').sort((a, b) => a.yPct - b.yPct);
            if (rail.length < 2) return [];
            const v: LayoutViolation[] = [];
            for (let i = 1; i < rail.length; i++) {
                const dy = rail[i].yPct - rail[i - 1].yPct;
                if (!nearGrid24(dy, 0.02)) {
                    v.push({
                        ruleId: 'rail-baseline',
                        severity: 'info',
                        message: `Rail block ${i} baseline drift`,
                        affectedRole: rail[i].role,
                    });
                }
            }
            return v;
        },
    },
    // Law 12 · Cover-line stack direction matches subject off-center
    {
        id: 'stack-direction',
        severity: 'info',
        check: (placements, g, preset) => {
            if (!g.subjectBox || preset.usesVerticalRail) return [];
            const subjCx = (g.subjectBox.x0 + g.subjectBox.x1) / 2;
            const feature = placements.find((p) => p.role === 'feature');
            if (!feature) return [];
            const expected = subjCx > 0.55 ? 'left' : subjCx < 0.45 ? 'right' : null;
            if (!expected) return [];
            const actual = feature.xPct < 0.5 ? 'left' : 'right';
            if (expected !== actual) {
                return [{
                    ruleId: 'stack-direction',
                    severity: 'info',
                    message: `Feature on ${actual}, subject suggests ${expected}`,
                    affectedRole: 'feature',
                }];
            }
            return [];
        },
    },
    // Law 13 · Special-issue treatment requires badge / border / frame
    {
        id: 'special-issue-marker',
        severity: 'info',
        check: (placements, _g, preset) => {
            if (preset.decorations.set === 'none') return [];
            // Verify a decoration shape exists for the style.
            const hasBadge = placements.some((p) => p.role === 'badge');
            const hasFrame = preset.decorations.frameColor !== null;
            if (preset.decorations.set === 'badge' && !hasBadge) {
                return [{
                    ruleId: 'special-issue-marker',
                    severity: 'info',
                    message: 'Style expects badge but none placed',
                }];
            }
            if ((preset.decorations.set === 'red-frame' ||
                preset.decorations.set === 'yellow-border') && !hasFrame) {
                return [{
                    ruleId: 'special-issue-marker',
                    severity: 'info',
                    message: 'Style expects border frame',
                }];
            }
            return [];
        },
    },
    // Law 14 · Plate colour inverts against photo luminance
    {
        id: 'plate-polarity',
        severity: 'warning',
        check: (placements, g, preset) => {
            const v: LayoutViolation[] = [];
            for (const p of placements) {
                if (!p.plate) continue;
                const expected = g.light ? '#000000' : '#FFFFFF';
                const actual = preset.colours.bodyColor;
                // Plate should be opposite polarity from text — they should differ.
                if (expected === actual) {
                    v.push({
                        ruleId: 'plate-polarity',
                        severity: 'warning',
                        message: `${p.role} plate polarity matches text colour`,
                        affectedRole: p.role,
                    });
                }
            }
            return v;
        },
    },
    // Law 15 · Crosshair decorations only in fragments style
    {
        id: 'crosshair-style-gate',
        severity: 'warning',
        check: (placements, _g, preset) => {
            const hasCrosshair = placements.some((p) => p.role === 'crosshair');
            if (hasCrosshair && !preset.usesCrosshairs) {
                return [{
                    ruleId: 'crosshair-style-gate',
                    severity: 'warning',
                    message: 'Crosshairs placed in non-fragments style',
                }];
            }
            return [];
        },
    },
];

/** Run all fifteen layout laws. Returns the violations (empty when legal). */
export const verifyLayout = (
    placements: SolvedPlacement[],
    g: GridAnalysis,
    preset: StylePreset,
): LayoutViolation[] => {
    const out: LayoutViolation[] = [];
    for (const law of LAYOUT_LAWS) {
        out.push(...law.check(placements, g, preset));
    }
    return out;
};

// ===========================================================================
// SECTION 7 · PUBLIC API — the solver entry point + apply-to-cover
// ---------------------------------------------------------------------------
// `solveEditorialCover` is the single entry point. It takes a measurement
// field, optional AI semantics, and an options bag (seed, explicit style).
// It returns the placements, decorations, violations, and chosen style.
// `applySolverToCover` then mutates a ComposedCover in place to write the
// solved geometry onto its text and shape specs.
// ===========================================================================

/** Pick a giant-type word for the HUMAN FRAGMENTS style. Falls back through
 *  semantics.subjectFocus → semantics.summary → a curated default word list
 *  keyed by the photo's hue/saturation vibe. */
const pickGiantWord = (
    g: GridAnalysis,
    semantics: SolverSemantics | null,
): string => {
    if (semantics?.subjectFocus && /^[A-Z\s]{3,12}$/i.test(semantics.subjectFocus)) {
        return semantics.subjectFocus.toUpperCase().split(/\s+/)[0];
    }
    // Vibe-keyed fallback word list — evocative, short, single-word.
    const warmWords = ['PULSE', 'TRACE', 'STEP', 'CRUISE', 'DRIFT', 'GLEAM', 'EMBER'];
    const coolWords = ['FORM', 'FRAME', 'PRESENCE', 'SPACE', 'STRATA', 'CIPHER'];
    const monoWords = ['FRAGMENT', 'INDEX', 'MARK', 'TRACE', 'PRESENCE'];
    if (g.highContrastMono) {
        return monoWords[Math.floor(g.hue / 60) % monoWords.length];
    }
    if (isWarmHue(g.hue)) {
        return warmWords[Math.floor(g.hue / 30) % warmWords.length];
    }
    return coolWords[Math.floor(g.hue / 30) % coolWords.length];
};

/** The solver entry point. Same photo + same seed → same cover.
 *  New click → new seed → the next-best optimal assignment. */
export function solveEditorialCover(
    g: GridAnalysis,
    semantics: SolverSemantics | null,
    opts: { seed?: number; style?: SystemStyleId } = {},
): SolverResult {
    const seed = (opts.seed ?? Math.floor(Math.random() * 0xffffffff)) >>> 0;
    const rng = mulberry32(seed);
    // ±6% weight noise per solve — gives k-best diversity across clicks
    // without changing the underlying cost function.
    const jitter = () => 1 + (rng() * 2 - 1) * 0.06;

    // Style resolution — explicit override → AI semantics → measurement.
    const style: SystemStyleId =
        opts.style ??
        (semantics?.genre && (KNOWN_STYLES as string[]).includes(semantics.genre)
            ? (semantics.genre as SystemStyleId)
            : gridVibe(g));
    const preset = getStylePreset(style);

    const zones = enumerateZones(g);

    // ── Build the cost matrix ───────────────────────────────────────────
    // Components are rows, zones are columns. We pad to square with phantom
    // columns so every component gets an assignment (infeasible = phantom).
    const cost: number[][] = COMPONENTS.map((comp) =>
        zones.map((z) => costOf(comp, z, g, preset, jitter)),
    );

    // Pad to square.
    const n = COMPONENTS.length;
    const square: number[][] = cost.map((row) => [...row]);
    while (square[0].length < n) {
        square.forEach((row) => row.push(INFEASIBLE * 2));
    }
    const assign = hungarian(square);

    // ── Resolve placements ──────────────────────────────────────────────
    const placements: SolvedPlacement[] = [];
    const byRole: Record<string, SolvedPlacement | undefined> = {};

    COMPONENTS.forEach((comp, i) => {
        const zi = assign[i];
        const z = zi >= 0 && zi < zones.length ? zones[zi] : null;
        if (!z || square[i][zi] >= INFEASIBLE) return;
        const p = resolvePlacement(comp, z, g, preset);
        placements.push(p);
        byRole[comp.id] = p;
    });

    // ── Style-specific geometry passes ─────────────────────────────────
    const decorations: DecorationShape[] = [];

    if (preset.usesVerticalRail) {
        // HUMAN FRAGMENTS: replace the standard feature/sub/teasers/dateline
        // placements with the structured vertical-rail stack.
        const rail = buildVerticalRail(preset, semantics);
        // Remove the standard placements for these roles.
        for (let i = placements.length - 1; i >= 0; i--) {
            if (['feature', 'sub', 'teasers', 'dateline'].includes(placements[i].role)) {
                placements.splice(i, 1);
            }
        }
        placements.push(...rail.placements);
        decorations.push(...rail.decorations);
        // Rebuild byRole.
        for (const p of placements) byRole[p.role] = p;
    }

    if (preset.usesGiantType) {
        // HUMAN FRAGMENTS: add the giant-type backdrop wordmark.
        const word = pickGiantWord(g, semantics);
        const giant = buildGiantType(preset, word);
        decorations.push(giant);
        // Update the masthead placement to reference the giant-type decoration.
        const m = byRole.masthead;
        if (m) {
            m.xPct = giant.xPct;
            m.yPct = giant.yPct;
            m.widthPct = giant.wPct * 100;
            m.heightPct = giant.hPct * 100;
            m.maskPhoto = giant.maskPhoto ?? false;
            m.depth = 'behind';
        }
    }

    // Crosshairs.
    const crosshairs = buildCrosshairs(preset);
    decorations.push(...crosshairs);
    // Also add them as placements so the layout-law engine can verify them.
    for (const ch of crosshairs) {
        placements.push({
            role: 'crosshair',
            zone: `E-${ch.xPct.toFixed(2)}-${ch.yPct.toFixed(2)}`,
            xPct: ch.xPct,
            yPct: ch.yPct,
            widthPct: ch.wPct,
            heightPct: ch.hPct,
            rotationDeg: 0,
            plate: false,
            depth: 'front',
            maskPhoto: false,
        });
    }

    // Accent block.
    const accent = buildAccentBlock(preset);
    if (accent) decorations.push(accent);

    // Border frame.
    const frame = buildBorderFrame(preset);
    if (frame) decorations.push(frame);

    // Plates for overlapping feature text.
    const feat = byRole.feature;
    if (feat && feat.plate) {
        decorations.push(buildPlate(feat, g.light));
    }

    // ── Sub / teasers follow feature in non-rail styles ────────────────
    if (!preset.usesVerticalRail) {
        const sub = byRole.sub;
        const teasers = byRole.teasers;
        if (feat && sub) {
            sub.xPct = feat.xPct;
            const subGap = feat.plate ? 0.088 : 0.065;
            sub.yPct = Math.round(clamp(feat.yPct + subGap, 0.1, 0.93) * 1000) / 1000;
            sub.plate = feat.plate;
        }
        if (feat && teasers) {
            teasers.xPct = sub ? sub.xPct : feat.xPct;
            teasers.yPct = Math.round(
                clamp((sub ? sub.yPct : feat.yPct) + 0.055, 0.12, 0.94) * 1000,
            ) / 1000;
        }
    }

    // ── Layout-law verification ─────────────────────────────────────────
    const violations = verifyLayout(placements, g, preset);

    return {
        placements,
        style,
        photoLight: g.light,
        decorations,
        violations,
        seed,
    };
}

// ===========================================================================
// SECTION 8 · APPLY TO COVER — rewrite the composed cover onto the geometry
// ---------------------------------------------------------------------------
// Mutates the composed cover in place: every text spec snaps to its solved
// zone; the feature gains a plate when the mathematics demands one; the
// barcode/QR moves to the solved corner; the giant-type backdrop, crosshair
// marks, accent block, and border frame are appended to the shapes list.
// ===========================================================================

const roleOfSpec = (spec: { id: string }): string | null => {
    const id = spec.id;
    if (id.includes('masthead')) return 'masthead';
    if (id.includes('dateline') || id.includes('rail')) return 'dateline';
    if (id.includes('feature')) return 'feature';
    if (id.includes('sub')) return 'sub';
    if (id.includes('teaser')) return 'teasers';
    if (id.includes('badge')) return 'badge';
    if (id.includes('crosshair')) return 'crosshair';
    return null;
};

/** Convert a DecorationShape into a SystemShapeSpec the composer can render. */
const decorationToShapeSpec = (d: DecorationShape): SystemShapeSpec => {
    const base = {
        id: `sys-decoration-${d.kind}-${Math.round(d.xPct * 1000)}-${Math.round(d.yPct * 1000)}`,
        xPct: d.xPct,
        yPct: d.yPct,
        wPct: d.wPct,
        hPct: d.hPct,
        rotationDeg: d.rotationDeg,
        opacity: 1,
        depth: d.depth,
    };
    switch (d.kind) {
        case 'frame':
            return {
                ...base,
                kind: 'rect',
                fillColor: d.color,
                fillOn: false,
                strokeColor: d.color,
                strokeW: d.thickness ?? 1.5,
                cornerPct: d.corner ?? 0,
            } as SystemShapeSpec;
        case 'crosshair':
            // A crosshair is two perpendicular thin rectangles.
            return {
                ...base,
                kind: 'rect',
                fillColor: d.color,
                fillOn: true,
                strokeColor: d.color,
                strokeW: 0,
                cornerPct: 0,
            } as SystemShapeSpec;
        case 'accent-block':
            return {
                ...base,
                kind: 'rect',
                fillColor: d.color,
                fillOn: true,
                strokeColor: d.color,
                strokeW: 0,
                cornerPct: 0,
            } as SystemShapeSpec;
        case 'em-dash':
            return {
                ...base,
                kind: 'rect',
                fillColor: d.color,
                fillOn: true,
                strokeColor: d.color,
                strokeW: 0,
                cornerPct: 0,
            } as SystemShapeSpec;
        case 'rail-bg':
            return {
                ...base,
                kind: 'rect',
                fillColor: d.color,
                fillOn: true,
                strokeColor: d.color,
                strokeW: 0,
                cornerPct: 0,
            } as SystemShapeSpec;
        case 'giant-type':
            // The giant-type is rendered as a text spec, not a shape — but
            // we expose it through the shape interface so the composer can
            // optionally treat it as a clipping mask for the photo.
            return {
                ...base,
                kind: 'rect',
                fillColor: d.color,
                fillOn: true,
                strokeColor: d.color,
                strokeW: 0,
                cornerPct: 0,
            } as SystemShapeSpec;
    }
};

/** Mutates the composed cover in place: snaps every text spec to its solved
 *  zone, adds plates where the solver demands them, moves the barcode to
 *  the solved corner, and appends all decoration shapes (frames, crosshairs,
 *  accent blocks, em-dashes, giant-type backdrops). */
export function applySolverToCover(cover: ComposedCover, solved: SolverResult): void {
    const byRole: Record<string, SolvedPlacement> = {};
    for (const p of solved.placements) byRole[p.role] = p;

    // ── HIERARCHY PASS — the v1 editorial ladder, restored ────────────
    // Two contracts rule this pass:
    //   1. UNITS — placements speak fractions (0-1), specs speak percent
    //      (0-100). Every write converts. (The old pass mixed the two,
    //      which slammed every solved element into the left edge.)
    //   2. SCALE — ROLES decide HOW BIG, zones decide WHERE. The preset's
    //      typeScale is the ladder: masthead towers, feature follows,
    //      subs and teasers shrink stepwise. A monotonic cascade (every
    //      rung ≥1.2× the rung below) guarantees the eye always reads
    //      masthead → feature → sub → teasers → micro, whatever the
    //      whitespace pockets looked like.
    const preset = getStylePreset(solved.style);
    const scaleOf = (role: string): number => {
        const s = (preset.typeScale as Record<string, number | undefined>)[role];
        return typeof s === 'number' ? s * 100 : 4;
    };
    const hDateline = Math.max(2, scaleOf('dateline'));
    const hTeasers = Math.max(scaleOf('teasers'), hDateline * 1.2);
    const hSub = Math.max(scaleOf('sub'), hTeasers * 1.2);
    const hFeature = Math.max(scaleOf('feature'), hSub * 1.25);
    const hMasthead = Math.max(scaleOf('masthead'), hFeature * 1.3);
    const ladder: Record<string, number> = {
        masthead: hMasthead,
        feature: hFeature,
        sub: hSub,
        teasers: hTeasers,
        dateline: hDateline,
    };
    const specsByRole: Record<string, SystemTextSpec[]> = {};
    for (const spec of cover.text as SystemTextSpec[]) {
        const role = roleOfSpec(spec);
        if (!role) continue;
        (specsByRole[role] ??= []).push(spec);
    }
    // MAGAZINE BANDS — the vertical grammar of every printed cover. The
    // whitespace solver picks the calm pocket, but only INSIDE the role's
    // band: dateline tops the page, the masthead lives in the upper field,
    // coverlines own the lower third. Without bands, a right-heavy subject
    //合法 parks every element in the one bottom-left pocket — the exact
    // corner cluster this pass existed to kill.
    const bandFor: Record<string, [number, number]> = {
        masthead: [0.14, 0.36],
        dateline: [0.05, 0.12],
        feature: [0.62, 0.86],
        sub: [0.66, 0.9],
        teasers: [0.6, 0.92],
        badge: [0.05, 0.16],
    };
    for (const [role, specs] of Object.entries(specsByRole)) {
        const p = byRole[role];
        if (!p) continue;
        // Vertical-rail placements carry their own baked hierarchy —
        // trust the rail's numbers for the rail's roles.
        const roleH =
            preset.usesVerticalRail && role !== 'masthead'
                ? p.heightPct
                : ladder[role] ?? p.heightPct;
        // Rail and giant-type geometry is deliberate full-canvas structure —
        // bands never touch it.
        const exempt =
            preset.usesVerticalRail ||
            (preset.usesGiantType && role === 'masthead') ||
            !!p.rotationDeg;
        const band = exempt ? null : bandFor[role] ?? null;
        const px = exempt ? p.xPct : clamp(p.xPct, 0.18, 0.82);
        const py = band ? clamp(p.yPct, band[0], band[1]) : p.yPct;
        specs.forEach((spec, k) => {
            spec.xPct = Math.round(clamp(px, 0.04, 0.96) * 1000) / 10;
            spec.yPct =
                Math.round(clamp(py + k * (roleH / 100 + 0.014), 0.04, 0.94) * 1000) / 10;
            spec.widthPct = p.widthPct;
            spec.heightPct =
                Math.round(Math.min(roleH, Math.max(2.2, p.heightPct * 1.3)) * 10) / 10;
            if (p.rotationDeg) spec.rotationDeg = p.rotationDeg;
            if (p.depth) spec.depth = p.depth;
        });
        // Giant-type styles render the masthead as the decoration wordmark
        // (the app expands it into a full TextLayer) — the composer's own
        // masthead spec would print the word twice, so it steps aside.
        if (preset.usesGiantType && role === 'masthead') {
            specs.forEach((spec) => {
                spec.opacity = 0;
            });
        }
    }
    // WIDTH hierarchy — the masthead is the widest statement on the page,
    // never narrower than the feature line beneath it.
    const mastSpec = specsByRole.masthead?.[0];
    const featW = specsByRole.feature?.[0]?.widthPct ?? 0;
    if (mastSpec && featW && mastSpec.widthPct < featW * 1.05) {
        mastSpec.widthPct = Math.min(92, Math.round(featW * 1.15));
    }

    // Plate the feature headline with a crisp high-contrast box — rail and
    // giant-type styles never plate the feature (their feature sits in the
    // vertical rail; plating it would crush the composition).
    const featSpec = (cover.text as SystemTextSpec[]).find(
        (s) => roleOfSpec(s) === 'feature',
    );
    if (featSpec && !preset.usesVerticalRail && !preset.usesGiantType) {
        const hasPlate = (cover.shapes as SystemShapeSpec[]).some(
            (s) =>
                s.kind === 'rect' &&
                s.fillOn &&
                Math.abs(s.xPct - featSpec.xPct) < 0.12 &&
                Math.abs(s.yPct - featSpec.yPct) < 0.12,
        );
        if (!hasPlate) {
            const textLen = (featSpec.text || '').trim().length;
            const approxTextWidthPct = Math.max(featSpec.widthPct, textLen * 3.4);
            const w = Math.min(94, approxTextWidthPct + 10);
            const h = featSpec.heightPct + 3.2;
            featSpec.widthPct = Math.max(featSpec.widthPct, approxTextWidthPct);
            featSpec.color = solved.photoLight ? '#FFFFFF' : '#000000';
            cover.shapes.push({
                id: 'sys-solver-plate-feat',
                kind: 'rect',
                xPct: featSpec.xPct,
                yPct: featSpec.yPct,
                wPct: w,
                hPct: h,
                fillColor: solved.photoLight ? '#000000' : '#FFFFFF',
                fillOn: true,
                strokeColor: solved.photoLight ? '#000000' : '#FFFFFF',
                strokeW: 0,
                cornerPct: 0,
                rotationDeg: 0,
                opacity: 1,
                depth: 'front',
            } as SystemShapeSpec);
        }
    }

    // Plate the sub headline with contrasting inverted box — percent-space
    // stacking below the feature plate; rail/giant styles are exempt (their
    // sub lives inside the vertical rail, not under a plated feature).
    const subSpec = (cover.text as SystemTextSpec[]).find(
        (s) => roleOfSpec(s) === 'sub',
    );
    if (subSpec && featSpec && !preset.usesVerticalRail && !preset.usesGiantType) {
        subSpec.yPct = Math.round(Math.min(93, featSpec.yPct + (featSpec.heightPct || 6) + 3.5) * 10) / 10;
        subSpec.xPct = featSpec.xPct;
        const textLen = (subSpec.text || '').trim().length;
        const approxTextWidthPct = Math.max(subSpec.widthPct, textLen * 2.8);
        const w = Math.min(90, approxTextWidthPct + 8);
        const h = subSpec.heightPct + 2.5;
        subSpec.widthPct = Math.max(subSpec.widthPct, approxTextWidthPct);
        subSpec.color = solved.photoLight ? '#000000' : '#FFFFFF';
        cover.shapes.push({
            id: 'sys-solver-plate-sub',
            kind: 'rect',
            xPct: subSpec.xPct,
            yPct: subSpec.yPct,
            wPct: w,
            hPct: h,
            fillColor: solved.photoLight ? '#FFFFFF' : '#000000',
            fillOn: true,
            strokeColor: solved.photoLight ? '#FFFFFF' : '#000000',
            strokeW: 0,
            cornerPct: 0,
            rotationDeg: 0,
            opacity: 1,
            depth: 'front',
        } as SystemShapeSpec);
    }

    // Move the scan element to the solved corner (fractions → percent).
    const scanP = byRole.scan;
    if (scanP) {
        const sc = (cover.shapes as SystemShapeSpec[]).find(
            (s) => s.kind === 'barcode' || s.kind === 'qr',
        );
        if (sc) {
            sc.xPct = Math.round(clamp(scanP.xPct, 0.08, 0.92) * 1000) / 10;
            sc.yPct = Math.round(clamp(scanP.yPct, 0.08, 0.92) * 1000) / 10;
        }
    }

    // Append all decorations (frames, crosshairs, accent blocks, etc.).
    for (const d of solved.decorations) {
        cover.shapes.push(decorationToShapeSpec(d));
    }
}

/** Convenience: solve a cover from a canvas in a single call. Returns the
 *  full SolverResult; the caller is responsible for applying it to a
 *  ComposedCover via `applySolverToCover`. */

/** Re-solve with a new seed — used by the "shuffle" UI affordance. The new
 *  seed produces the next-best optimal assignment (k-best diversity). */
export function reshuffle(
    g: GridAnalysis,
    semantics: SolverSemantics | null,
    prev: SolverResult,
): SolverResult {
    let nextSeed = (prev.seed + 1) >>> 0;
    // A few tries to find a meaningfully different placement set.
    let best: SolverResult = prev;
    let bestDiff = 0;
    for (let i = 0; i < 5; i++) {
        const candidate = solveEditorialCover(g, semantics, {
            seed: nextSeed,
            style: prev.style,
        });
        // Count how many placements differ from the previous solve.
        let diff = 0;
        for (const p of candidate.placements) {
            const prevP = prev.placements.find((q) => q.role === p.role);
            if (!prevP) continue;
            if (Math.abs(prevP.xPct - p.xPct) > 0.02 || Math.abs(prevP.yPct - p.yPct) > 0.02) {
                diff++;
            }
        }
        if (diff > bestDiff) {
            best = candidate;
            bestDiff = diff;
        }
        if (diff >= 2) break;
        nextSeed = (nextSeed + 0x9e3779b9) >>> 0;
    }
    return best;
}

/** Validate a composed cover against the layout-law engine — useful for
 *  the UI to surface warnings before the user exports. */
export function validateCover(
    cover: ComposedCover,
    g: GridAnalysis,
    style: SystemStyleId,
): LayoutViolation[] {
    const preset = getStylePreset(style);
    const placements: SolvedPlacement[] = [];
    for (const spec of cover.text as SystemTextSpec[]) {
        const role = roleOfSpec(spec);
        if (!role) continue;
        placements.push({
            role,
            zone: '',
            xPct: spec.xPct,
            yPct: spec.yPct,
            widthPct: spec.widthPct,
            heightPct: spec.heightPct ?? 5,
            rotationDeg: spec.rotationDeg ?? 0,
            plate: false,
            depth: spec.depth ?? 'front',
            maskPhoto: false,
        });
    }
    return verifyLayout(placements, g, preset);
}

/** Returns true when the solver can produce a legal placement set for the
 *  given field and style — i.e. at least one assignment has zero critical
 *  violations. Useful for filtering the style dropdown to feasible options. */
export function isStyleFeasible(
    g: GridAnalysis,
    style: SystemStyleId,
    semantics: SolverSemantics | null = null,
): boolean {
    const result = solveEditorialCover(g, semantics, { style });
    return !result.violations.some((v) => v.severity === 'critical');
}

/** Compute a per-style feasibility score (0..1) for a given field — useful
 *  for surfacing the top three recommended styles in the UI. */
export function styleRecommendations(
    g: GridAnalysis,
    semantics: SolverSemantics | null = null,
): Array<{ style: SystemStyleId; score: number; feasible: boolean }> {
    const out: Array<{ style: SystemStyleId; score: number; feasible: boolean }> = [];
    for (const preset of STYLE_PRESETS) {
        const score = preset.matches(g);
        const feasible = isStyleFeasible(g, preset.id, semantics);
        out.push({ style: preset.id, score, feasible });
    }
    out.sort((a, b) => b.score - a.score);
    return out;
}

/** Returns the full cost matrix that would be fed to the Hungarian for a
 *  given field and style. Useful for debugging "why did the solver pick
 *  that zone?" questions. */
export function debugCostMatrix(
    g: GridAnalysis,
    style: SystemStyleId,
): number[][] {
    const preset = getStylePreset(style);
    const zones = enumerateZones(g);
    const rng = mulberry32(0); // deterministic for debug
    const jitter = () => 1;
    return COMPONENTS.map((comp) =>
        zones.map((z) => costOf(comp, z, g, preset, jitter)),
    );
}

    return enumerateZones(g);
}

    return getStylePreset(id);

    return {

/** Deterministic 32-bit RNG — same seed, same stream, same cover. Local
 *  copy so the solver stays pure (no imports from the composer's rng). */
const mulberry32 = (seed: number) => () => {
    seed |= 0;
    seed = (seed + 0x6d2b79f5) | 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
};

const clamp = (v: number, lo: number, hi: number): number => Math.max(lo, Math.min(hi, v));
const clamp01 = (v: number): number => clamp(v, 0, 1);
const lerp = (a: number, b: number, t: number): number => a + (b - a) * t;

/** Hermite smoothstep — the only easing the geometry passes need. */
const smoothstep = (edge0: number, edge1: number, x: number): number => {
    const t = clamp01((x - edge0) / (edge1 - edge0 || 1e-6));
    return t * t * (3 - 2 * t);
};

/** Fractional-space rectangle. THE geometry currency of this file. */
interface Rect {
    x0: number;
    y0: number;
    x1: number;
    y1: number;
}

const rectOf = (x0: number, y0: number, x1: number, y1: number): Rect => ({ x0, y0, x1, y1 });
const rectW = (r: Rect): number => r.x1 - r.x0;
const rectH = (r: Rect): number => r.y1 - r.y0;
const rectCx = (r: Rect): number => (r.x0 + r.x1) / 2;
const rectCy = (r: Rect): number => (r.y0 + r.y1) / 2;
const rectArea = (r: Rect): number => Math.max(1e-6, rectW(r) * rectH(r));

const rectIntersect = (a: Rect, b: Rect): Rect | null => {
    const x0 = Math.max(a.x0, b.x0);
    const y0 = Math.max(a.y0, b.y0);
    const x1 = Math.min(a.x1, b.x1);
    const y1 = Math.min(a.y1, b.y1);
    return x1 > x0 && y1 > y0 ? rectOf(x0, y0, x1, y1) : null;
};

const rectUnion = (a: Rect, b: Rect): Rect =>
    rectOf(Math.min(a.x0, b.x0), Math.min(a.y0, b.y0), Math.max(a.x1, b.x1), Math.max(a.y1, b.y1));

const rectExpand = (r: Rect, padX: number, padY: number): Rect =>
    rectOf(r.x0 - padX, r.y0 - padY, r.x1 + padX, r.y1 + padY);

/** Overlap of a over b, as a fraction of b's area — Gen II's proven metric. */
const overlapRatio = (a: Rect, b: Rect): number => {
    const ix = Math.max(0, Math.min(a.x1, b.x1) - Math.max(a.x0, b.x0));
    const iy = Math.max(0, Math.min(a.y1, b.y1) - Math.max(a.y0, b.y0));
    return (ix * iy) / rectArea(b);
};

/** Intersection-over-union — the dedupe metric for the zone engine. */
const iou = (a: Rect, b: Rect): number => {
    const inter = rectIntersect(a, b);
    if (!inter) return 0;
    const union = rectArea(rectUnion(a, b));
    return rectArea(inter) / union;
};

/** WCAG relative luminance of an 8-bit channel triple. */
const relLuminance = (r01: number, g01: number, b01: number): number => {
    const ch = (c: number) => (c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4));
    return 0.2126 * ch(r01) + 0.7152 * ch(g01) + 0.0722 * ch(b01);
};

/** WCAG contrast ratio between two luminances (1..21). White-on-black = 21. */
const contrastRatio = (lumA: number, lumB: number): number => {
    const hi = Math.max(lumA, lumB);
    const lo = Math.min(lumA, lumB);
    return (hi + 0.05) / (lo + 0.05);
};

/** Ink luminance extremes used across the engine. */
const INK_LUM = 0.04; // near-black ink
const PAPER_LUM = 0.97; // near-white ink

/**
 * THE UNIT-GRAMMAR GUARD. In dev builds only, screams if a placement or
 * decoration violates the house law: positions in 0–1, sizes in 0–100.
 * This check exists because an entire afternoon once died to a ×100.
 */
const DEV = typeof process !== 'undefined' && typeof process.env !== 'undefined' && process.env.NODE_ENV !== 'production';
const devGuardUnits = (label: string, x: number, y: number, w: number, h: number): void => {
    if (!DEV) return;
    if (x < -0.02 || x > 1.02 || y < -0.02 || y > 1.02) {
        console.warn(`[solver:units] ${label}: xPct/yPct must be FRACTIONS 0–1 — got (${x}, ${y})`);
    }
    if (w < -0.5 || w > 100.5 || h < -0.5 || h > 100.5) {
        console.warn(`[solver:units] ${label}: wPct/hPct must be PERCENT 0–100 — got (${w}, ${h})`);
    }
};

// ===========================================================================
// §1 · THE MEASUREMENT FIELD — one 96×120 downsample becomes a 24×30 cell
//      grid of luminance, texture energy, saturation and calmness, plus a
//      trimmed subject box, a horizon line, and a distance-to-busyness map
//      that tells the whitespace engine how PROTECTED every cell is.
// ===========================================================================

export const SOLVER_GRID = { cols: 24, rows: 30 } as const;

export interface GridAnalysis {
    cols: number;
    rows: number;
    /** per-cell mean luminance 0-1 */
    lum: Float32Array;
    /** per-cell texture/edge energy 0-1 (gradient magnitude) */
    energy: Float32Array;
    /** per-cell saturation 0-1 */
    sat: Float32Array;
    /** per-cell calmness 0-1 (1 - normalized energy) */
    calm: Float32Array;
    /** per-cell distance to the nearest busy cell, in cell units, 0-8 */
    shelter: Float32Array;
    /** overall photo reads light (plate polarity) */
    light: boolean;
    /** bounding box of the visual mass (normalized, percentile-trimmed), null if diffuse */
    subjectBox: Rect | null;
    /** a tighter core of the subject — where the face/mass really lives */
    subjectCore: Rect | null;
    /** row (0-1) where horizontal structure peaks — the horizon */
    horizonY: number;
    /** circular-mean hue 0-360 and mean saturation — the colour vibe */
    hue: number;
    satAvg: number;
    /** column marginal calmness (length cols) */
    colProfile: Float32Array;
    /** row marginal calmness (length rows) */
    rowProfile: Float32Array;
}

export function analyzeGridFromCanvas(cv: HTMLCanvasElement): GridAnalysis | null {
    try {
        const ctx = cv.getContext('2d', { willReadFrequently: true });
        if (!ctx) return null;
        const W0 = cv.width;
        const H0 = cv.height;
        if (!W0 || !H0) return null;

        // FAST FIRST STEP — downsample to a 96×120 thumbnail before ANY
        // pixel read. A full-resolution getImageData allocates tens of
        // megabytes and stalls the frame; the thumbnail is ~100× cheaper
        // and just as accurate for a 24×30 field (4×4 px per cell).
        const W = 96;
        const H = 120;
        const off = document.createElement('canvas');
        off.width = W;
        off.height = H;
        const octx = off.getContext('2d', { willReadFrequently: true });
        if (!octx) return null;
        octx.drawImage(cv, 0, 0, W, H);

        const { cols, rows } = SOLVER_GRID;
        const n = cols * rows;
        const lum = new Float32Array(n);
        const sat = new Float32Array(n);
        const rs = new Float64Array(n);
        const gs = new Float64Array(n);
        const bs = new Float64Array(n);
        const cnt = new Float64Array(n);

        const d = octx.getImageData(0, 0, W, H).data;
        // Pixel pass — accumulate RGB into cells (one pass over the tiny
        // thumbnail; sub-millisecond by construction).
        for (let i = 0; i < d.length; i += 4) {
            const px = (i / 4) | 0;
            const cxs = px % W;
            const cys = (px / W) | 0;
            const c = Math.min(cols - 1, ((cxs / W) * cols) | 0);
            const r = Math.min(rows - 1, ((cys / H) * rows) | 0);
            const k = r * cols + c;
            rs[k] += d[i];
            gs[k] += d[i + 1];
            bs[k] += d[i + 2];
            cnt[k]++;
        }
        let sinH = 0;
        let cosH = 0;
        let satSum = 0;
        let lumSum = 0;
        for (let k = 0; k < n; k++) {
            const c = Math.max(1, cnt[k]);
            const r = rs[k] / c / 255;
            const g = gs[k] / c / 255;
            const b = bs[k] / c / 255;
            const l = 0.299 * r + 0.587 * g + 0.114 * b;
            lum[k] = l;
            const max = Math.max(r, g, b);
            const min = Math.min(r, g, b);
            const s = max <= 0 ? 0 : (max - min) / max;
            sat[k] = s;
            lumSum += l;
            satSum += s;
            if (max - min > 1e-4) {
                let h: number;
                if (max === r) h = ((g - b) / (max - min)) % 6;
                else if (max === g) h = (b - r) / (max - min) + 2;
                else h = (r - g) / (max - min) + 4;
                h *= 60;
                if (h < 0) h += 360;
                sinH += Math.sin((h * Math.PI) / 180);
                cosH += Math.cos((h * Math.PI) / 180);
            }
        }

        // Energy pass — gradient magnitude over the cell lattice (a cheap
        // Sobel on the downsampled field; edges = subject, flat = space).
        const energy = new Float32Array(n);
        for (let r = 0; r < rows; r++) {
            for (let c = 0; c < cols; c++) {
                const k = r * cols + c;
                const l = lum[k];
                const lu = lum[Math.max(0, r - 1) * cols + c];
                const ld = lum[Math.min(rows - 1, r + 1) * cols + c];
                const ll = lum[r * cols + Math.max(0, c - 1)];
                const lr = lum[r * cols + Math.min(cols - 1, c + 1)];
                energy[k] = clamp01((Math.abs(lu - ld) + Math.abs(ll - lr)) / 2 / 0.16);
            }
        }
        const calm = new Float32Array(n);
        for (let k = 0; k < n; k++) calm[k] = clamp01(1 - energy[k]);

        // Subject statistics — mean and spread of busyness for trimming.
        let eMean = 0;
        for (let k = 0; k < n; k++) eMean += energy[k];
        eMean /= n;
        let eVar = 0;
        for (let k = 0; k < n; k++) eVar += (energy[k] - eMean) * (energy[k] - eMean);
        const eStd = Math.sqrt(eVar / n);

        // PERCENTILE-TRIMMED SUBJECT BOX — Gen II took every cell above a
        // threshold, so one busy corner dragged the box across the whole
        // page and the adjacency math flattened. Gen III collects hit
        // coordinates, sorts them, and trims the outer 8% per edge: the
        // box hugs the mass that actually matters.
        const thr = eMean + 0.55 * eStd;
        const hitXs: number[] = [];
        const hitYs: number[] = [];
        for (let r = 0; r < rows; r++) {
            for (let c = 0; c < cols; c++) {
                if (energy[r * cols + c] > thr) {
                    hitXs.push((c + 0.5) / cols);
                    hitYs.push((r + 0.5) / rows);
                }
            }
        }
        const quantile = (arr: number[], q: number): number => {
            if (!arr.length) return 0;
            const s = [...arr].sort((a, b) => a - b);
            return s[clamp(Math.floor(q * (s.length - 1)), 0, s.length - 1)];
        };
        let subjectBox: Rect | null = null;
        let subjectCore: Rect | null = null;
        if (hitXs.length >= 5) {
            subjectBox = rectOf(
                clamp(quantile(hitXs, 0.08), 0, 1),
                clamp(quantile(hitYs, 0.08), 0, 1),
                clamp(quantile(hitXs, 0.92), 0, 1),
                clamp(quantile(hitYs, 0.92), 0, 1),
            );
            subjectCore = rectOf(
                clamp(quantile(hitXs, 0.3), 0, 1),
                clamp(quantile(hitYs, 0.3), 0, 1),
                clamp(quantile(hitXs, 0.7), 0, 1),
                clamp(quantile(hitYs, 0.7), 0, 1),
            );
        }

        // HORIZON — the row whose horizontal energy run is strongest. A
        // landscape's text should respect the ground; a portrait's should
        // respect the shoulder line. Cheap: max over rows of smoothed
        // row energy, kept only when it clearly dominates.
        let horizonY = 0.5;
        let bestRun = 0;
        let secondRun = 0;
        for (let r = 2; r < rows - 2; r++) {
            let run = 0;
            for (let c = 1; c < cols - 1; c++) {
                run += Math.abs(lum[r * cols + c + 1] - lum[r * cols + c - 1]);
            }
            if (run > bestRun) {
                secondRun = bestRun;
                bestRun = run;
                horizonY = r / rows;
            } else if (run > secondRun) {
                secondRun = run;
            }
        }
        if (bestRun < secondRun * 1.6) horizonY = 0.5; // no clear horizon — stay neutral

        // SHELTER MAP — bounded BFS from every busy cell: how far each cell
        // sits from the nearest structure. Whitespace deep inside a calm
        // region is PREMIUM (a masthead there never collides); calm cells
        // hugging the subject are cushion space (good for plating, risky
        // for bare ink). Capped at 8 — beyond that it is all the same
        // "deep space" to the cost function.
        const shelter = new Float32Array(n).fill(8);
        const queue: number[] = [];
        for (let k = 0; k < n; k++) {
            if (energy[k] > thr) {
                shelter[k] = 0;
                queue.push(k);
            }
        }
        let head = 0;
        while (head < queue.length) {
            const k = queue[head++];
            const r = (k / cols) | 0;
            const c = k % cols;
            const dcur = shelter[k];
            if (dcur >= 8) continue;
            for (let dr = -1; dr <= 1; dr++) {
                for (let dc = -1; dc <= 1; dc++) {
                    const rr = r + dr;
                    const cc = c + dc;
                    if (rr < 0 || rr >= rows || cc < 0 || cc >= cols) continue;
                    const kk = rr * cols + cc;
                    if (shelter[kk] > dcur + 1) {
                        shelter[kk] = dcur + 1;
                        queue.push(kk);
                    }
                }
            }
        }

        // Marginals — the column/row whitespace profile (interior only).
        const colProfile = new Float32Array(cols);
        const rowProfile = new Float32Array(rows);
        for (let c = 0; c < cols; c++) {
            let s = 0;
            for (let r = 2; r < rows - 2; r++) s += calm[r * cols + c];
            colProfile[c] = s / (rows - 4);
        }
        for (let r = 0; r < rows; r++) {
            let s = 0;
            for (let c = 2; c < cols - 2; c++) s += calm[r * cols + c];
            rowProfile[r] = s / (cols - 4);
        }

        let hue = Math.atan2(sinH, cosH) * (180 / Math.PI);
        if (hue < 0) hue += 360;
        return {
            cols,
            rows,
            lum,
            energy,
            sat,
            calm,
            shelter,
            light: lumSum / Math.max(1, n) > 0.58,
            subjectBox,
            subjectCore,
            horizonY,
            hue,
            satAvg: satSum / Math.max(1, n),
            colProfile,
            rowProfile,
        };
    } catch {
        return null;
    }
}

/** Derive the composer's PhotoAnalysis from the grid — ONE scan feeds
 *  both engines, so ASSEMBLE's opening step is a single 96×120 pass. */
export function gridToPhotoAnalysis(g: GridAnalysis): PhotoAnalysis {
    const bandCalm = (x0: number, y0: number, x1: number, y1: number): number => {
        let s = 0;
        let n = 0;
        for (let r = 0; r < g.rows; r++) {
            for (let c = 0; c < g.cols; c++) {
                const cc = (c + 0.5) / g.cols;
                const rr = (r + 0.5) / g.rows;
                if (cc < x0 || cc > x1 || rr < y0 || rr > y1) continue;
                s += g.calm[r * g.cols + c];
                n++;
            }
        }
        return Math.round((s / Math.max(1, n)) * 100) / 100;
    };
    const core = g.subjectCore ?? g.subjectBox;
    const cx = core ? rectCx(core) : 0.5;
    return {
        headroom: bandCalm(0, 0, 1, 0.18),
        leftSpace: bandCalm(0, 0.18, 0.24, 0.72),
        rightSpace: bandCalm(0.76, 0.18, 1, 0.72),
        bottomSpace: bandCalm(0, 0.74, 1, 0.94),
        photoLight: g.light,
        subjectX: cx < 0.42 ? 'left' : cx > 0.58 ? 'right' : 'center',
        hue: Math.round(g.hue),
        sat: Math.round(g.satAvg * 100) / 100,
    } as PhotoAnalysis;
}

// ===========================================================================
// §2 · THE WHITESPACE ENGINE — where Gen III earns its commission. The 17
//      Gen II lattice zones survive (they encode editorial positions that
//      no algorithm would invent but every art director recognizes), and
//      three new families join them:
//
//        · MAXIMAL CALM RECTANGLES — grown from calm seeds like largest-
//          empty-rectangles over the busy-cell map: the true whitespace
//          the photo actually offers, wherever it happens to be.
//        · SUBJECT-ADJACENT CUSHIONS — bands hugging the trimmed subject
//          box: the zones covers use for plating and sandwiching.
//        · HORIZON BANDS — the calm strip above/below the horizon line.
//
//      Everything funnels through non-max suppression (IoU < 0.45) and a
//      scored ranking capped at MAX_ZONES. No zone is imagined; every
//      rectangle is measured calm, flat, sheltered and luminant.
// ===========================================================================

const MAX_ZONES = 64;

interface Zone {
    id: string;
    x0: number;
    y0: number;
    x1: number;
    y1: number;
    w: number;
    h: number;
    cx: number;
    cy: number;
    calm: number;
    flat: number;
    lum: number;
    /** mean shelter (distance to structure) in cell units, 0-8 */
    shelter: number;
    /** overlap fraction with the trimmed subject box */
    subject: number;
    tags: string[];
    /** composite quality score, higher = better whitespace */
    score: number;
}

type Grid = GridAnalysis;

const zoneMetrics = (g: Grid, id: string, x0: number, y0: number, x1: number, y1: number, tags: string[]): Zone => {
    x0 = clamp01(x0);
    y0 = clamp01(y0);
    x1 = clamp01(x1);
    y1 = clamp01(y1);
    let calmSum = 0;
    let lumSum = 0;
    let shelterSum = 0;
    let n = 0;
    for (let r = 0; r < g.rows; r++) {
        for (let c = 0; c < g.cols; c++) {
            const cc = (c + 0.5) / g.cols;
            const rr = (r + 0.5) / g.rows;
            if (cc < x0 || cc > x1 || rr < y0 || rr > y1) continue;
            const k = r * g.cols + c;
            calmSum += g.calm[k];
            lumSum += g.lum[k];
            shelterSum += g.shelter[k];
            n++;
        }
    }
    n = Math.max(1, n);
    const lumMean = lumSum / n;
    let varAcc = 0;
    for (let r = 0; r < g.rows; r++) {
        for (let c = 0; c < g.cols; c++) {
            const cc = (c + 0.5) / g.cols;
            const rr = (r + 0.5) / g.rows;
            if (cc < x0 || cc > x1 || rr < y0 || rr > y1) continue;
            const dl = g.lum[r * g.cols + c] - lumMean;
            varAcc += dl * dl;
        }
    }
    const rect = rectOf(x0, y0, x1, y1);
    return {
        id,
        x0,
        y0,
        x1,
        y1,
        w: x1 - x0,
        h: y1 - y0,
        cx: (x0 + x1) / 2,
        cy: (y0 + y1) / 2,
        calm: calmSum / n,
        flat: clamp01(1 - Math.sqrt(varAcc / n) * 3),
        lum: lumMean,
        shelter: shelterSum / n,
        subject: g.subjectBox ? overlapRatio(rect, g.subjectBox) : 0,
        tags,
        score: 0,
    };
};

/** The Gen II lattice — kept verbatim in geometry, rescored by the new
 *  engine. These are the positions art directors draw by hand. */
const latticeZones = (g: Grid): Zone[] => [
    zoneMetrics(g, 'T1', 0.06, 0.045, 0.94, 0.15, ['top', 'band']),

/**
 * MAXIMAL CALM RECTANGLES — the discovery pass. Walks the calm mask on a
 * stride, grows each seed right and down while calmness holds, then
 * trims to the measured interior. This is how the engine finds the real
 * whitespace of THIS photo: the sky patch nobody planned for, the calm
 * wall on the left, the empty studio floor. Bounded: ≤ 40 seeds × O(cols
 * + rows) growth.
 */
const maximalCalmRectangles = (g: Grid): Zone[] => {
    const { cols, rows, calm } = g;
    const CALM_SEED = 0.62;
    const CALM_HOLD = 0.52;
    const out: Zone[] = [];
    const cellRect = (c0: number, r0: number, c1: number, r1: number): Zone | null => {
        const x0 = c0 / cols;
        const y0 = r0 / rows;
        const x1 = (c1 + 1) / cols;
        const y1 = (r1 + 1) / rows;
        if (x1 - x0 < 0.16 || y1 - y0 < 0.08) return null; // too small to typeset in
        return zoneMetrics(g, `X${out.length}`, x0, y0, x1, y1, ['discovered']);
    };
    const colCalm = (c: number, r0: number, r1: number): boolean => {
        let s = 0;
        for (let r = r0; r <= r1; r++) s += calm[r * cols + c];
        return s / Math.max(1, r1 - r0 + 1) >= CALM_HOLD;
    };
    const rowCalm = (r: number, c0: number, c1: number): boolean => {
        let s = 0;
        for (let c = c0; c <= c1; c++) s += calm[r * cols + c];
        return s / Math.max(1, c1 - c0 + 1) >= CALM_HOLD;
    };
    let made = 0;
    for (let r0 = 1; r0 < rows - 1 && made < 40; r0 += 3) {
        for (let c0 = 1; c0 < cols - 1 && made < 40; c0 += 3) {
            if (calm[r0 * cols + c0] < CALM_SEED) continue;
            let c1 = c0;
            let r1 = r0;
            // Greedy two-axis growth: extend width while columns stay calm,
            // then height while rows stay calm; repeat until stable.
            for (let guard = 0; guard < 8; guard++) {
                let grew = false;
                while (c1 + 1 < cols - 1 && colCalm(c1 + 1, r0, r1)) {
                    c1++;
                    grew = true;
                }
                while (r1 + 1 < rows - 1 && rowCalm(r1 + 1, c0, c1)) {
                    r1++;
                    grew = true;
                }
                if (!grew) break;
            }
            const z = cellRect(c0, r0, c1, r1);
            if (z) {
                out.push(z);
                made++;
                // Skip past this rectangle's row band — seeds inside it
                // would only rediscover the same whitespace.
                r0 = Math.min(rows - 2, r1);
                break;
            }
        }
    }
    return out;
};

/**
 * SUBJECT-ADJACENT CUSHIONS — bands hugging the trimmed subject box. The
 * coverline stack that points AT the subject lives in the shoulder cushion;
 * the plated statement that crosses it starts from the flank cushions.
 * Only emitted when the subject box is real and leaves room.
 */
const subjectCushions = (g: Grid): Zone[] => {
    const s = g.subjectBox;
    if (!s) return [];
    const out: Zone[] = [];
    const pad = 0.03;
    if (s.x0 > 0.24) {
        out.push(zoneMetrics(g, `S-L`, pad, clamp(s.y0 - 0.02, 0.05, 1), s.x0 - 0.02, clamp(s.y1 + 0.02, 0, 0.95), ['cushion', 'left', 'mid']));
    }
    if (s.x1 < 0.76) {
        out.push(zoneMetrics(g, `S-R`, clamp(s.x1 + 0.02, 0.05, 1), clamp(s.y0 - 0.02, 0.05, 1), 1 - pad, clamp(s.y1 + 0.02, 0, 0.95), ['cushion', 'right', 'mid']));
    }
    if (s.y0 > 0.24) {
        out.push(zoneMetrics(g, `S-T`, 0.06, 0.05, 0.94, s.y0 - 0.02, ['cushion', 'top', 'band']));
    }
    if (s.y1 < 0.74) {
        out.push(zoneMetrics(g, `S-B`, 0.06, clamp(s.y1 + 0.02, 0.05, 1), 0.94, 0.95, ['cushion', 'bottom', 'band']));
    }
    return out;
};

/**
 * HORIZON BANDS — the calm strips above and below the strongest
 * horizontal structure. When a photo has a real horizon, type respects
 * it the way a landscape architect respects a waterline.
 */
const horizonBands = (g: Grid): Zone[] => {
    if (g.horizonY === 0.5) return []; // neutral — no clear horizon
    const out: Zone[] = [];
    if (g.horizonY > 0.34) {
        out.push(zoneMetrics(g, 'H-A', 0.08, 0.06, 0.92, clamp(g.horizonY - 0.03, 0.14, 0.9), ['horizon', 'top', 'band']));
    }
    if (g.horizonY < 0.66) {
        out.push(zoneMetrics(g, 'H-B', 0.08, clamp(g.horizonY + 0.03, 0.1, 0.86), 0.92, 0.94, ['horizon', 'bottom', 'band']));
    }
    return out;
};

/**
 * ZONE SCORING — the composite whitespace quality. Weights are editorial
 * judgment encoded once, in daylight:
 *   calm       2.6 — stillness is the raw material of legibility
 *   shelter    1.1 — deep space never collides with the subject
 *   flat       0.8 — tonal evenness keeps ink honest corner to corner
 *   thirds     0.6 — anchors on the optical power grid read as designed
 *   edge-guard 0.9 — zones drifting off-canvas are discounted
 */
const THIRD_POINTS_X = [1 / 3, 2 / 3];
const scoreZone = (z: Zone): number => {
    let s = 2.6 * z.calm + 1.1 * clamp01(z.shelter / 5) + 0.8 * z.flat;
    const nearThirdsX = THIRD_POINTS_X.some((p) => Math.abs(z.cx - p) < 0.09);
    const nearThirdsY = [1 / 3, 2 / 3].some((p) => Math.abs(z.cy - p) < 0.09);
    if (nearThirdsX && nearThirdsY) s += 0.6;
    else if (nearThirdsX || nearThirdsY) s += 0.3;
    if (z.x0 < 0.02 || z.x1 > 0.98 || z.y0 < 0.02 || z.y1 > 0.98) s -= 0.9;
    return s;
};

/** Non-max suppression: drop any zone whose IoU with a better-scored
 *  keeper exceeds 0.45. Lattice zones get a +0.25 heritage bonus — a
 *  hand-position beats a discovered one when they tie. */
const nonMaxSuppress = (zones: Zone[]): Zone[] => {
    const ranked = zones
        .map((z) => ({ z, s: scoreZone(z) + (z.tags.includes('discovered') ? 0 : 0.25) }))
        .sort((a, b) => b.s - a.s);
    const kept: Zone[] = [];
    for (const { z, s } of ranked) {
        let ok = true;
        for (const k of kept) {
            if (iou(rectOf(z.x0, z.y0, z.x1, z.y1), rectOf(k.x0, k.y0, k.x1, k.y1)) > 0.45) {
                ok = false;
                break;
            }
        }
        if (ok) {
            z.score = s;
            kept.push(z);
        }
        if (kept.length >= MAX_ZONES) break;
    }
    return kept;
};

const enumerateZones = (g: Grid): Zone[] =>
    nonMaxSuppress([...latticeZones(g), ...maximalCalmRectangles(g), ...subjectCushions(g), ...horizonBands(g)]);

// ===========================================================================
// §3 · SEMANTICS & COPY FITNESS — the AI's only jurisdiction. It describes
//      the photograph and (optionally) supplies the words; the engine
//      measures whether those words FIT the geometry it is about to
//      choose, before choosing it. Copy that cannot fit its zone at a
//      dignified size is a layout bug wearing a writer's hat.
// ===========================================================================

export interface SolverSemantics {
    summary?: string;
    genre?: string;
    mood?: string[];
    subjectFocus?: string;
}

/** The editorial roles the engine places. Order = hierarchy rank. */
export type SolverRole = 'masthead' | 'dateline' | 'feature' | 'sub' | 'teasers' | 'badge' | 'scan';

const ROLE_RANK: Record<string, number> = {
    masthead: 0,
    feature: 1,
    sub: 2,
    teasers: 3,
    dateline: 4,
    badge: 5,
    scan: 6,
};

/** Optional real copy, so fitness is measured on the actual words. */
export type SolverCopy = Partial<Record<SolverRole, string>>;

/**
 * Width a single line of `chars` characters occupies at a given cap height,
 * in PERCENT of canvas width. Average editorial glyph advance ≈ 0.62 of
 * cap height; tracking adds em-fraction per character. This estimate is
 * the bridge between the writer's word count and the geometer's column.
 */
const estimateLineWidthPct = (chars: number, heightPct: number, trackingEm = 0): number =>
    chars * (0.62 + trackingEm) * heightPct;

/**
 * How the copy fits a zone: the scale factor (≤ 1) at which the longest
 * line fits the zone width, and the number of lines a wrapped block needs.
 * Returns null when no dignified fit exists (scale < 0.55 — the words are
 * simply too many for this whitespace; the cost matrix will feel it).
 */
const fitCopyToZone = (
    text: string,
    zoneWidthPct: number,
    heightPct: number,
    trackingEm = 0,
): { scale: number; lines: number } | null => {
    const rawLines = text.split('\n').filter((l) => l.trim().length > 0);
    if (!rawLines.length) return { scale: 1, lines: 1 };
    const longest = Math.max(...rawLines.map((l) => l.trim().length));
    const need = estimateLineWidthPct(longest, heightPct, trackingEm);
    const scale = clamp01(zoneWidthPct / Math.max(1e-6, need));
    if (scale < 0.55) return null;
    // Wrapped line count: total glyphs over what one zone line carries.
    const perLine = Math.max(4, Math.floor((longest * zoneWidthPct) / Math.max(1e-6, need)));
    const total = rawLines.reduce((acc, l) => acc + Math.max(1, Math.ceil(l.trim().length / perLine)), 0);
    return { scale, lines: clamp(total, 1, 6) };
};

/** The giant wordmark seed: the most typograph-able strong word in the
 *  semantics — 3–8 letters, alphabetic, punchy. */
const giantWordFrom = (semantics: SolverSemantics | null, copy: SolverCopy): string => {
    const src = `${copy.masthead ?? ''} ${semantics?.subjectFocus ?? ''} ${semantics?.summary ?? ''}`;
    const word = src
        .split(/\s+/)
        .map((w) => w.replace(/[^A-Za-z]/g, '').toUpperCase())
        .find((w) => w.length >= 3 && w.length <= 8);
    return word || 'PULSE';
};

// ===========================================================================
// §4 · THE TYPE LADDER — hierarchy is solved, never inherited. The style's
//      class sets the base rungs; the chosen zone's capacity scales them;
//      a monotonicity pass enforces the visual law: every rung steps down
//      by a clear ratio, masthead to teaser, or the ladder is wrong.
// ===========================================================================

export interface TypeLadder {
    masthead: number;
    feature: number;
    sub: number;
    teaser: number;
    dateline: number;
    /** the breath gap between stack rungs, as a fraction of canvas height */
    breath: number;
}

/** Style classes — the two dials that actually differ between houses. */
interface StyleClass {
    /** base masthead cap height, PERCENT */
    masthead: number;
    /** feature rung when it leads a stack vs when it is a plated statement */
    featureStack: number;
    featureStatement: number;
    /** global tracking character */
    tracking: number;
}

const CLASSIC_CLASS: StyleClass = {
    masthead: 16,
    featureStack: 3.4,
    featureStatement: 5.2,
    tracking: 0.08,
};

const DISPLAY_CLASS: StyleClass = {
    masthead: 18,
    featureStack: 4.6,
    featureStatement: 7,
    tracking: -0.02,
};

const SERIF_CLASS: StyleClass = {
    masthead: 15,
    featureStack: 3.2,
    featureStatement: 4.8,
    tracking: 0.1,
};

const styleClassOf = (style: string): StyleClass => {
    switch (style) {
        case 'vogue':
        case 'bazaar':
            return SERIF_CLASS;
        case 'zine':
        case 'dazed':
        case 'cyber':
            return DISPLAY_CLASS;
        default:
            return CLASSIC_CLASS;
    }
};

/**
 * Solve the ladder for a chosen cluster zone and signature.
 * `zoneH` is the cluster zone's height as a FRACTION (0–1); the stack's
 * total ink must fit inside it with breath gaps, and every rung must
 * step down from its neighbour by at least the monotonic ratios below.
 */
const LADDER_MONO = { mastheadOverFeature: 2.1, featureOverSub: 1.45, subOverTeaser: 1.12 };

const solveTypeLadder = (
    style: string,
    clusterZoneH: number,
    statement: boolean,
    overrides: Partial<Record<SolverRole, number>> | undefined,
): TypeLadder => {
    const cls = styleClassOf(style);
    let feature = statement ? cls.featureStatement : cls.featureStack;
    let sub = feature * 0.55;
    let teaser = sub * 0.82;
    const masthead = cls.masthead;
    const dateline = 1.9;

    // Capacity: the stack (feature + sub + teaser rows) plus breaths must
    // fit 88% of the zone's height. If it cannot, scale the whole stack
    // down together — hierarchy is ratio, not absolute size.
    const breath = 0.022;
    const stackH = (feature + sub + teaser) / 100 + 2 * breath;
    const capacity = clamp(clusterZoneH, 0.12, 0.9) * 0.88;
    if (stackH > capacity) {
        const k = capacity / stackH;
        feature *= k;
        sub *= k;
        teaser *= k;
    } else if (stackH < capacity * 0.72) {
        // GESTALT · SCALE EXAGGERATION — "if it doesn't stand out,
        // exaggerate the scale." The down-fit above is only half the
        // law: a stack that fits timidly inside generous whitespace
        // reads PLAIN. When the zone affords headroom (stack under
        // 72% of capacity), the ladder GROWS to claim 86% of it —
        // bounded by the class's statement ceiling so quiet genres
        // never shout past their register. The masthead follows via
        // the return-line dominance law, so the WHOLE ladder towers.
        const ceiling = statement ? cls.featureStatement : cls.featureStack;
        const k = Math.min((capacity * 0.86) / stackH, ceiling / feature);
        feature *= k;
        sub *= k;
        teaser *= k;
    }

    // Monotonicity — the law, enforced after all other forces.
    sub = Math.min(sub, feature / LADDER_MONO.featureOverSub);
    teaser = Math.min(teaser, sub / LADDER_MONO.subOverTeaser);
    feature = Math.max(feature, sub * LADDER_MONO.featureOverSub);
    if (overrides) {
        if (overrides.feature != null) feature = overrides.feature;
        if (overrides.sub != null) sub = Math.min(overrides.sub, feature / LADDER_MONO.featureOverSub);
        if (overrides.teasers != null) teaser = Math.min(overrides.teasers, sub / LADDER_MONO.subOverTeaser);
        if (overrides.masthead != null) {
            // A preset may tower the masthead, but never below the law.
            const floor = feature * LADDER_MONO.mastheadOverFeature;
            const wanted = overrides.masthead;
            // feature could exceed `floor` if presets forced a small masthead;
            // in that case feature yields instead — the brand always wins.
            if (wanted < floor) feature = Math.min(feature, wanted / LADDER_MONO.mastheadOverFeature);
        }
    }

    return {
        masthead: Math.max(overrides?.masthead ?? masthead, feature * LADDER_MONO.mastheadOverFeature),
        feature: Math.round(feature * 100) / 100,
        sub: Math.round(sub * 100) / 100,
        teaser: Math.round(teaser * 100) / 100,
        dateline,
        breath,
    };
};

// ===========================================================================
// §5 · THE ASSIGNMENT OPTIMIZER — the Gen II Hungarian core, verbatim
//      (it is exact, unbeatable, and battle-proven), plus k-best restarts:
//      seeded cost perturbations produce genuinely different optima, and
//      a distinctness filter keeps only layouts that MOVE.
// ===========================================================================

const INFEASIBLE = 1e6;

/** Jonker-style Hungarian assignment with potentials. cost[r][c], returns
 *  col index per row (minimizes total cost). Classic, exact, unbeatable. */
const hungarian = (cost: number[][]): number[] => {
    const n = cost.length;
    const INF = 1e15;
    const u = new Array<number>(n + 1).fill(0);
    const v = new Array<number>(n + 1).fill(0);
    const p = new Array<number>(n + 1).fill(0);
    const way = new Array<number>(n + 1).fill(0);
    for (let i = 1; i <= n; i++) {
        p[0] = i;
        let j0 = 0;
        const minv = new Array<number>(n + 1).fill(INF);
        const used = new Array<boolean>(n + 1).fill(false);
        do {
            used[j0] = true;
            const i0 = p[j0];
            let delta = INF;
            let j1 = 0;
            for (let j = 1; j <= n; j++) {
                if (used[j]) continue;
                const cur = cost[i0 - 1][j - 1] - u[i0] - v[j];
                if (cur < minv[j]) {
                    minv[j] = cur;
                    way[j] = j0;
                }
                if (minv[j] < delta) {
                    delta = minv[j];
                    j1 = j;
                }
            }
            for (let j = 0; j <= n; j++) {
                if (used[j]) {
                    u[p[j]] += delta;
                    v[j] -= delta;
                } else {
                    minv[j] -= delta;
                }
            }
            j0 = j1;
        } while (p[j0] !== 0);
        do {
            const j1 = way[j0];
            p[j0] = p[j1];
            j0 = j1;
        } while (j0);
    }
    const rowToCol = new Array<number>(n).fill(-1);
    for (let j = 1; j <= n; j++) if (p[j] > 0) rowToCol[p[j] - 1] = j - 1;
    return rowToCol;
};

interface KBestResult {
    assignment: number[];
    totalCost: number;
}

/**
 * k-best by perturbation restart: solve once on the true cost, then re-
 * solve on seeded jittered copies. Each candidate keeps its TRUE cost;
 * dedupe requires a mean per-role displacement of at least `minShift`
 * zone-index distance so "different" means different. Returns best-first.
 */
const kBestAssignments = (cost: number[][], k: number, rng: () => number, minShift = 1.4): KBestResult[] => {
    const trueTotal = (a: number[]): number =>
        a.reduce((acc, col, row) => (col >= 0 && cost[row][col] < INFEASIBLE ? acc + cost[row][col] : acc + INFEASIBLE), 0);
    const seen: { assignment: number[]; totalCost: number }[] = [];
    const push = (a: number[]): void => {
        if (a.some((c) => c < 0)) return;
        if (seen.some((s) => s.assignment.every((c, i) => c === a[i]))) return;
        seen.push({ assignment: [...a], totalCost: trueTotal(a) });
    };
    push(hungarian(cost));
    for (let attempt = 0; attempt < k * 3 && seen.length < k; attempt++) {
        const jittered = cost.map((row) => row.map((c) => (c >= INFEASIBLE ? c : c * (1 + (rng() * 2 - 1) * 0.35))));
        const a = hungarian(jittered);
        if (a.some((c) => c < 0)) continue;
        const far = seen.every((s) => {
            const shift = a.reduce((acc, col, i) => acc + Math.abs(col - s.assignment[i]), 0) / a.length;
            return shift >= minShift;
        });
        if (far) push(a);
    }
    return seen.sort((a, b) => a.totalCost - b.totalCost);
};

// ===========================================================================
// §6 · THE SOLVER — cluster-unit assignment. The feature/sub/teaser stack
//      travels as ONE body (Gen II assigned them separately and stacked
//      afterthought-fashion — the bunched-cover bug). Here the cluster's
//      zone must have the CAPACITY for the whole ladder, and the roles are
//      distributed inside it with breath gaps. Quadrant spread is checked
//      after the optimum; a lopsided page re-solves with balance bonuses.
// ===========================================================================

export interface SolvedPlacement {
    role: string;
    zone: string;
    xPct: number;
    yPct: number;
    widthPct: number;
    rotationDeg: number;
    plate: boolean;
    depth: 'behind' | 'front';
    /** solved type rung (heightPct, PERCENT 0-100) — the ladder's word */
    heightPct?: number;
    /** ink polarity the physics chose for this placement */
    polarity?: 'ink' | 'plate';
    /** measured mean luminance under this placement (0-1) */
    zoneLum?: number;
}

export interface SolverResult {
    placements: SolvedPlacement[];
    style: SolverStyleToken;
    photoLight: boolean;
    decorations: DecorationShape[];
    violations: LayoutViolation[];
    seed: number;
    ladder: TypeLadder;
    /** k-best runners-up — hand these to "next layout" for instant variety */
    alternates: SolvedPlacement[][];
}

const KNOWN_STYLES: SystemStyleId[] = [
    'vogue', 'zine', 'kinfolk', 'noir', 'gq', 'dazed', 'natgeo', 'cyber', 'vinyl', 'architectural',
];

/** The colour-mood → genre map, on solver measurements (AI-free fallback). */
const gridVibe = (g: Grid): SystemStyleId => {
    const { hue, satAvg: sat, light } = g;
    if (sat < 0.16) return light ? 'kinfolk' : 'noir';
    const warm = hue >= 340 || (hue >= 12 && hue <= 50);
    if (warm && sat >= 0.34) return 'vogue'; // warm+saturated appetite → couture (the food genre is post-HEAD)
    if (hue >= 70 && hue < 170) return light ? 'kinfolk' : 'vinyl';
    if (hue >= 170 && hue < 265) return light ? 'natgeo' : 'cyber';
    if (hue >= 265 && hue < 340 && sat >= 0.25) return 'dazed';
    return light ? 'kinfolk' : 'gq';
};

/** Snap to the 24-division optical grid when within tolerance — the quiet
 *  mechanical rhythm that makes print layouts feel set, not dropped. */
const opticalSnap = (v: number): number => {
    const s = Math.round(v * 24) / 24;
    return Math.abs(s - v) < 0.011 ? s : v;
};

const quadrantOf = (x: number, y: number): 0 | 1 | 2 | 3 => {
    const top = y < 0.5;
    const left = x < 0.5;
    return top ? (left ? 0 : 1) : left ? 2 : 3;
};

export function solveEditorialCover(
    g: Grid,
    semantics: SolverSemantics | null,
    opts: { seed?: number; style?: SolverStyleToken; copy?: SolverCopy } = {},
): SolverResult {
    const seed = (opts.seed ?? Math.floor(Math.random() * 0xffffffff)) >>> 0;
    const rng = mulberry32(seed);
    const style: SolverStyleToken =
        opts.style ??
        (semantics?.genre && (KNOWN_STYLES as string[]).includes(semantics.genre) ? (semantics.genre as SystemStyleId) : gridVibe(g));
    const signature = resolveSignature(style);
    const copy = opts.copy ?? {};
    const jitter = () => 1 + (rng() * 2 - 1) * 0.06; // ±6% weight noise → k-best diversity

    const zones = enumerateZones(g);
    const subjectCx = g.subjectCore ? rectCx(g.subjectCore) : g.subjectBox ? rectCx(g.subjectBox) : 0.5;
    // The text mass lives in the CALMER half; the scan corner sits opposite.
    let leftCalm = 0;
    let rightCalm = 0;
    for (let c = 0; c < 6; c++) leftCalm += g.colProfile[c];
    for (let c = g.cols - 6; c < g.cols; c++) rightCalm += g.colProfile[c];
    const textSide: 'left' | 'right' = leftCalm >= rightCalm ? 'left' : 'right';

    // ── Cluster units ─────────────────────────────────────────────────────
    // masthead · dateline · CLUSTER(feature+sub+teasers) · badge · scan.
    // The cluster is one row of the matrix; its internal distribution is
    // §4/§6 geometry, run after the optimum is chosen.
    const UNITS: { id: string; aspect: number; minCalm: number }[] = [
        { id: 'masthead', aspect: 3.4, minCalm: 0.4 },
        { id: 'dateline', aspect: 9.0, minCalm: 0.48 },
        { id: 'cluster', aspect: 3.0, minCalm: 0.42 },
        { id: 'badge', aspect: 5.0, minCalm: 0.42 },
        { id: 'scan', aspect: 2.2, minCalm: 0.25 },
    ];

    // Copy fitness needs a provisional ladder; statement-ness depends on
    // the zone, but the scale differs little — compute one ladder with the
    // page-wide calm as proxy capacity, refine inside the geometry pass.
    const provisional = solveTypeLadder(style, 0.5, false, signature.typeScale);

    const buildCost = (spreadBonus: number[]): number[][] =>
        UNITS.map((unit) =>
            zones.map((z, zi) => {
                if (z.calm < unit.minCalm) return INFEASIBLE;
                const zoneAspect = z.w / Math.max(0.02, z.h);
                let c =
                    2.3 * (1 - z.calm) * jitter() +
                    0.8 * Math.abs(Math.log(zoneAspect / unit.aspect)) * jitter() +
                    0.35 * (1 - z.flat) +
                    0.5 * (1 - clamp01(z.shelter / 5)) +
                    spreadBonus[zi] * jitter();
                switch (unit.id) {
                    case 'masthead':
                        if (z.cy > 0.62) c += 0.9; // masthead leads the upper two-thirds
                        if (g.subjectBox && overlapRatio(z, g.subjectBox) > 0.5) c += 0.35; // sandwich legal, not preferred
                        if (signature.giant) c -= 0.3; // giant wordmark styles want the field
                        break;
                    case 'dateline':
                        if (z.cy > 0.16 && z.cy < 0.9) c += 0.8;
                        if (z.cy > 0.5) c += 0.3;
                        break;
                    case 'cluster': {
                        c += 0.9 * Math.abs(z.cx - subjectCx); // coverlines point AT the subject
                        if (z.cy < 0.3) c += 0.4;
                        // CAPACITY — the whole ladder stack must fit the zone.
                        const stackNeed = (provisional.feature + provisional.sub + provisional.teaser) / 100 + 2 * provisional.breath;
                        if (stackNeed > z.h * 0.92) c += 1.4 * (stackNeed / Math.max(0.05, z.h * 0.92) - 1);
                        // COPY FITNESS — the actual words, if we have them.
                        const featCopy = copy.feature;
                        if (featCopy) {
                            const fit = fitCopyToZone(featCopy, z.w * 100, provisional.feature, styleClassOf(style).tracking);
                            if (!fit) c += 1.2;
                            else c += 0.5 * (1 - fit.scale);
                        }
                        break;
                    }
                    case 'badge':
                        if (!z.tags.includes('corner')) c += INFEASIBLE;
                        break;
                    case 'scan': {
                        if (!z.tags.includes('corner')) c += INFEASIBLE;
                        else if (textSide === 'left' && z.tags.includes('left')) c += 0.8;
                        else if (textSide === 'right' && z.tags.includes('right')) c += 0.8;
                        break;
                    }
                }
                return c;
            }),
        );

    // Pad to square with phantom zones so every unit gets an answer.
    const n = UNITS.length;
    const padSquare = (c: number[][]): number[][] => {
        const sq = c.map((row) => [...row]);
        while (sq[0].length < n) sq.forEach((row) => row.push(INFEASIBLE * 2));
        return sq;
    };

    // ── the optimum (and its runners-up) ─────────────────────────────────
    let cost = buildCost(zones.map(() => 0));
    let cands = kBestAssignments(padSquare(cost), 4, rng);
    if (!cands.length) {
        // Degenerate photo (nothing calm enough): fall back to infeasible-
        // tolerant costs so the composer defaults still rule the page.
        cost = cost.map((row) => row.map((c) => (c >= INFEASIBLE ? 60 : c)));
        cands = kBestAssignments(padSquare(cost), 1, rng);
    }

    /** Turn an assignment into concrete placements + the final ladder.
     *  Pure with respect to its inputs — the winner and every alternate
     *  run through the same geometry, no special cases. */
    const materialize = (assign: number[]): { placements: SolvedPlacement[]; ladder: TypeLadder } => {
        const placements: SolvedPlacement[] = [];
        const zoneOf = (i: number): Zone | null => {
            const zi = assign[i];
            return zi >= 0 && zi < zones.length ? zones[zi] : null;
        };
        const zoneLumOf = (z: Zone | null): number => (z ? z.lum : g.light ? 0.8 : 0.2);

        // -- masthead ------------------------------------------------------
        const mastZ = zoneOf(0);
        if (mastZ && cost[0][assign[0]] < INFEASIBLE) {
            const rail = mastZ.tags.includes('column') && mastZ.h / Math.max(0.04, mastZ.w) >= 1.9;
            let widthPct: number;
            let rotationDeg = 0;
            if (rail) {
                // Vertical masthead: text length runs down the column height.
                widthPct = clamp(mastZ.h * 100 * 0.85, 46, 64);
                rotationDeg = mastZ.tags.includes('left') ? -90 : 90;
            } else {
                widthPct = clamp(mastZ.w * 100 * 0.92, 46, 92);
            }
            const cx = rail
                ? clamp(mastZ.x0 + (mastZ.tags.includes('left') ? 0.045 : -0.045) + (mastZ.tags.includes('left') ? 0.0 : mastZ.w), 0.03, 0.97)
                : opticalSnap(clamp(mastZ.cx, 0.06, 0.94));
            const cy = opticalSnap(clamp(mastZ.cy, 0.08, 0.92));
            const sandwich = !!g.subjectBox && overlapRatio(mastZ, g.subjectBox) > 0.4 && !rail;
            placements.push({
                role: 'masthead',
                zone: mastZ.id,
                xPct: Math.round(cx * 1000) / 1000,
                yPct: Math.round(cy * 1000) / 1000,
                widthPct: Math.round(widthPct),
                rotationDeg,
                plate: false,
                depth: sandwich ? 'behind' : 'front',
                zoneLum: zoneLumOf(mastZ),
                polarity: 'ink',
            });
        }

        // -- dateline --------------------------------------------------------
        const dateZ = zoneOf(1);
        if (dateZ && cost[1][assign[1]] < INFEASIBLE) {
            placements.push({
                role: 'dateline',
                zone: dateZ.id,
                xPct: Math.round(opticalSnap(clamp(dateZ.cx, 0.08, 0.92)) * 1000) / 1000,
                yPct: Math.round(opticalSnap(clamp(dateZ.cy, 0.06, 0.14)) * 1000) / 1000,
                widthPct: Math.round(clamp(dateZ.w * 100 * 0.9, 30, 84)),
                rotationDeg: 0,
                plate: false,
                depth: 'front',
                zoneLum: zoneLumOf(dateZ),
                polarity: 'ink',
            });
        }

        // -- the cluster: feature / sub / teasers, distributed in-zone --------
        const clZ = zoneOf(2);
        if (clZ && cost[2][assign[2]] < INFEASIBLE) {
            const statement = !!g.subjectBox && overlapRatio(clZ, g.subjectBox) > 0.35 && clZ.calm < 0.55;
            const ladder = solveTypeLadder(style, clZ.h, statement, signature.typeScale);
            const stackX = opticalSnap(clamp(clZ.cx, 0.12, 0.88));
            const widthPct = Math.round(clamp(clZ.w * 100 * 0.92, 24, 60));

            // Distribution: the stack starts one rung below the zone's top
            // and each following role lands after the previous rung plus a
            // breath gap — never a fixed offset, always proportional, so
            // the cluster fills its whitespace without ever bunching.
            const top = clamp(clZ.y0 + ladder.feature / 200 + 0.012, 0.08, 0.9);
            const featP: SolvedPlacement = {
                role: 'feature',
                zone: clZ.id,
                xPct: Math.round(stackX * 1000) / 1000,
                yPct: Math.round(opticalSnap(top) * 1000) / 1000,
                widthPct,
                rotationDeg: 0,
                plate: statement,
                depth: 'front',
                heightPct: ladder.feature,
                zoneLum: zoneLumOf(clZ),
                polarity: statement ? 'plate' : 'ink',
            };
            placements.push(featP);

            const subY = clamp(top + ladder.feature / 200 + ladder.breath + ladder.sub / 200, 0.1, 0.93);
            const subP: SolvedPlacement = {
                role: 'sub',
                zone: clZ.id,
                xPct: featP.xPct,
                yPct: Math.round(opticalSnap(subY) * 1000) / 1000,
                widthPct: Math.max(20, widthPct - 4),
                rotationDeg: 0,
                plate: statement,
                depth: 'front',
                heightPct: ladder.sub,
                zoneLum: zoneLumOf(clZ),
                polarity: statement ? 'plate' : 'ink',
            };
            placements.push(subP);

            const teasY = clamp(subY + ladder.sub / 200 + ladder.breath + ladder.teaser / 200, 0.12, 0.94);
            placements.push({
                role: 'teasers',
                zone: clZ.id,
                xPct: featP.xPct,
                yPct: Math.round(opticalSnap(teasY) * 1000) / 1000,
                widthPct: Math.max(18, widthPct - 8),
                rotationDeg: 0,
                plate: false,
                depth: 'front',
                heightPct: ladder.teaser,
                zoneLum: zoneLumOf(clZ),
                polarity: 'ink',
            });

            // Ink physics — bare ink needs contrast ≥ 3.2 against its zone;
            // otherwise the rung demands a plate (the app's plating law
            // re-cuts the rect; we only rule on necessity).
            if (!statement) {
                const inkLum = g.light ? PAPER_LUM : INK_LUM;
                if (contrastRatio(inkLum, zoneLumOf(clZ)) < 3.2) {
                    featP.plate = true;
                    featP.polarity = 'plate';
                    subP.plate = true;
                    subP.polarity = 'plate';
                }
            }

            return { placements, ladder };
        }

        // Cluster infeasible — composer defaults carry the coverline stack.
        return { placements, ladder: provisional };
    };

    // ── spread balance: the page must breathe in more than one corner ----
    const quadrantMass = (ps: SolvedPlacement[]): number[] => {
        const m = [0, 0, 0, 0];
        for (const p of ps) {
            const w = p.role === 'masthead' ? 2 : p.role === 'feature' ? 3 : p.role === 'sub' || p.role === 'teasers' ? 2 : 1;
            m[quadrantOf(p.xPct, p.yPct)] += w;
        }
        return m;
    };
    let winner = cands[0];
    let mat = materialize(winner.assignment);
    {
        const m = quadrantMass(mat.placements);
        const total = m.reduce((a, b) => a + b, 0) || 1;
        if (Math.max(...m) / total > 0.7) {
            // Lopsided — re-solve with the crowded quadrant's zones taxed.
            const crowded = m.indexOf(Math.max(...m));
            const spreadBonus = zones.map((z) => (quadrantOf(z.cx, z.cy) === crowded ? 0.9 : 0));
            const rebalanced = kBestAssignments(padSquare(buildCost(spreadBonus)), 2, rng);
            if (rebalanced.length) {
                winner = rebalanced[0];
                mat = materialize(winner.assignment);
            }
        }
    }

    const alternates: SolvedPlacement[][] = [];
    for (const cand of cands.slice(1)) {
        if (cand.assignment.every((c, i) => c === winner.assignment[i])) continue;
        alternates.push(materialize(cand.assignment).placements);
        if (alternates.length >= 2) break;
    }

    // ── the masthead dominance invariant (today's lesson, made law) ------
    // In band zones the masthead's column visibly dominates the feature's:
    // "brand towers, coverline answers". Vertical rails and giant-wordmark
    // styles are exempt — their width grammar is different on purpose.
    const mast = mat.placements.find((p) => p.role === 'masthead');
    const feat = mat.placements.find((p) => p.role === 'feature');
    if (mast && feat && !mast.rotationDeg && !signature.giant && /^(T|M|C|D|S|H|X)/.test(mast.zone)) {
        if (mast.widthPct < feat.widthPct + 8) {
            mast.widthPct = Math.min(92, feat.widthPct + 8);
        }
    }
    for (const p of mat.placements) {
        devGuardUnits(`placement:${p.role}`, p.xPct, p.yPct, p.widthPct, p.heightPct ?? 0);
    }

    const decorations = emitDecorations(signature, {
        g,
        word: giantWordFrom(semantics, copy),
        ladder: mat.ladder,
    });
    for (const d of decorations) {
        devGuardUnits(`decoration:${d.kind}`, d.xPct, d.yPct, d.wPct, d.hPct);
    }

    const violations = verifyLaws(mat.placements, mat.ladder, g, signature);

    return {
        placements: mat.placements,
        style,
        photoLight: g.light,
        decorations,
        violations,
        seed,
        ladder: mat.ladder,
        alternates,
    };
}

/** The composition the PHOTOGRAPH asks for (§4 decision logic, §7 crop
 *  inference, §8 quadrant rules) — measured, never guessed. */
export interface CompositionPlan {
    /** §4 — masthead/subject relationship the negative space supports */
    mastheadRel: 'front' | 'behind' | 'side';
    /** §5/§8 — where the coverline stack wants to live */
    stackSide: 'left' | 'right' | 'both' | 'bottom' | 'top';
    /** §7 — the subject crop the photo already is */
    crop: 'full-bleed' | 'three-quarter' | 'head-shoulders' | 'closeup' | 'vignette' | 'off-center';
    /** measured calm of the band above the subject's head (0–1) */
    headroom: number;
    /** measured calm of the outer left / right strips (0–1) */
    leftStrip: number;
    rightStrip: number;
    /** §8 — quadrants flat enough to own typography */
    empty: Array<'T' | 'B' | 'L' | 'R'>;
}

const meanOf = (a: Float32Array, from: number, to: number): number => {
    const i0 = Math.max(0, Math.floor(from * a.length));
    const i1 = Math.min(a.length, Math.ceil(to * a.length));
    if (i1 <= i0) return 0;
    let s = 0;
    for (let i = i0; i < i1; i++) s += a[i];
    return s / (i1 - i0);
};

/** §4's decision tree, on solver measurements:
 *  headroom? → masthead in front, in the top band (4.1).
 *  side strip? → whitespace strategy, masthead in the strip (4.3).
 *  neither? → overlay and let the plate physics carry legibility. */
export function inferComposition(g: Grid): CompositionPlan {
    const subjTop = g.subjectBox?.y0 ?? 0.3;
    const headroom = subjTop > 0.06 ? meanOf(g.rowProfile, 0, Math.min(subjTop, 0.35)) : 0.08;
    const leftStrip = meanOf(g.colProfile, 0, 0.24);
    const rightStrip = meanOf(g.colProfile, 0.76, 1);
    const empty: CompositionPlan['empty'] = [];
    if (meanOf(g.rowProfile, 0, 0.5) > 0.55) empty.push('T');
    if (meanOf(g.rowProfile, 0.5, 1) > 0.55) empty.push('B');
    if (meanOf(g.colProfile, 0, 0.5) > 0.55) empty.push('L');
    if (meanOf(g.colProfile, 0.5, 1) > 0.55) empty.push('R');

    const mastheadRel: CompositionPlan['mastheadRel'] =
        headroom > 0.5 ? 'front' : Math.max(leftStrip, rightStrip) > 0.55 ? 'side' : 'front';

    let stackSide: CompositionPlan['stackSide'];
    const L = empty.includes('L');
    const R = empty.includes('R');
    if (L && R) stackSide = 'both';
    else if (L) stackSide = 'left';
    else if (R) stackSide = 'right';
    else if (empty.includes('B')) stackSide = 'bottom';
    else if (empty.includes('T')) stackSide = 'top';
    else stackSide = 'bottom'; // dense photo: the §5.5 fallback strip

    const b = g.subjectBox;
    const bw = b ? b.x1 - b.x0 : 1;
    const bh = b ? b.y1 - b.y0 : 1;
    let crop: CompositionPlan['crop'] = 'head-shoulders';
    if (!b) crop = 'vignette';
    else if (bw < 0.38 && bh > 0.5) crop = 'closeup';
    else if (bh > 0.92) crop = 'full-bleed';
    else if (Math.abs((b.x0 + b.x1) / 2 - 0.5) > 0.14) crop = 'off-center';
    else if (bh > 0.62) crop = 'three-quarter';

    return { mastheadRel, stackSide, crop, headroom, leftStrip, rightStrip, empty };
}

/** §2 geometry — the brand band each masthead position owns. Units:
 *  x/y fractions, widthPct PERCENT — the house grammar. */
export interface MastheadBand {
    x: number;
    y: number;
    widthPct: number;
    rotationDeg: number;
    depth: 'front' | 'behind';
}

export function mastheadBand(pos: MastheadPosition, plan: CompositionPlan): MastheadBand {
    const side: 'left' | 'right' = plan.leftStrip >= plan.rightStrip ? 'left' : 'right';
    switch (pos) {
        case 'A':
            // Full-width over the subject — but §4.3: when there is no
            // headroom and a strip is calm, the whitespace strategy
            // moves the wordmark INTO the strip beside the subject.
            if (plan.headroom < 0.5 && Math.max(plan.leftStrip, plan.rightStrip) > 0.55) {
                return {
                    x: side === 'left' ? 0.27 : 0.73,
                    y: 0.115,
                    widthPct: 44,
                    rotationDeg: 0,
                    depth: 'front',
                };
            }
            return { x: 0.5, y: 0.095, widthPct: 92, rotationDeg: 0, depth: 'front' };
        case 'B':
            // The celebrity override — the wordmark sits BEHIND the
            // subject's head; legibility ≥70% is the plate system's job.
            return { x: 0.5, y: 0.095, widthPct: 92, rotationDeg: 0, depth: 'behind' };
        case 'C':
            // The news nameplate — compact, centered, newspaper voice.
            return { x: 0.5, y: 0.075, widthPct: 40, rotationDeg: 0, depth: 'front' };
        case 'D':
            // The indie corner anchor.
            return { x: 0.24, y: 0.08, widthPct: 44, rotationDeg: 0, depth: 'front' };
        case 'E':
            // The side-rail vertical — the whole top goes to the photo.
            return {
                x: side === 'left' ? 0.055 : 0.945,
                y: 0.5,
                widthPct: 58,
                rotationDeg: side === 'left' ? -90 : 90,
                depth: 'front',
            };
        case 'F':
            // The inverted commemorative — brand at the floor.
            return { x: 0.5, y: 0.925, widthPct: 70, rotationDeg: 0, depth: 'front' };
    }
}

// §7 · STYLE SIGNATURES — the good parts of Gen IIb, as DATA. A signature

/** Runtime preset registry — `preset:<name>` tokens live here. */
const PRESET_SIGNATURES = new Map<string, StyleSignature>();

/**
 * Register a preset signature at runtime. No engine edit, no recompile.
 *
 *   registerStyleSignature('preset:fragments', {
 *       giant: { heightPct: 88, widthPct: 74, maskPhoto: true },
 *       rail: { x: 0.875, widthPct: 18, dividers: true },
 *       crosshairs: { sizePct: 3 },
 *       accent: '#D94B28',
 *   });
 *
 * Throws on malformed tokens (must be `preset:` + a non-empty name) so a
 * typo'd preset fails at registration, never at layout time.
 */
export function registerStyleSignature(token: SolverStyleToken, signature: StyleSignature): void {
    if (!token.startsWith('preset:') || token.length <= 'preset:'.length) {
        throw new Error(`[solver] preset tokens must look like "preset:name" — got "${token}"`);
    }
    PRESET_SIGNATURES.set(token.slice('preset:'.length), signature);
}

/** Everything the engine can currently style — genres first, presets after. */
export function listStyleSignatures(): SolverStyleToken[] {
    return [
        ...(Object.keys(SIGNATURES) as SystemStyleId[]),
        ...[...PRESET_SIGNATURES.keys()].map((name) => `preset:${name}` as SolverStyleToken),
    ];
}

const normalizeGiant = (v: boolean | GiantSpec | undefined): GiantSpec | null =>
    typeof v === 'object' ? v : v ? { widthPct: 92, heightPct: 90, opacity: 0.14, maskPhoto: true, weight: 900, tracking: -0.03 } : null;
const normalizeRail = (v: boolean | RailSpec | undefined): RailSpec | null =>
    typeof v === 'object' ? v : v ? { x: 0.875, widthPct: 18, dividers: true, backdrop: false } : null;
const normalizeCross = (v: boolean | CrosshairSpec | undefined): CrosshairSpec | null =>
    typeof v === 'object' ? v : v ? { positions: [[0.08, 0.08], [0.92, 0.92]], sizePct: 3, arm: 0.018 } : null;
const normalizeAccent = (v: string | AccentSpec | undefined): AccentSpec | null =>
    v == null ? null : typeof v === 'string' ? { color: v, xPct: 0.24, yPct: 0.5, wPct: 12, hPct: 8 } : v;

const resolveSignature = (token: SolverStyleToken): NormalizedSignature => {
    const raw: StyleSignature | undefined = token.startsWith('preset:')
        ? PRESET_SIGNATURES.get(token.slice('preset:'.length))
        : SIGNATURES[token as SystemStyleId];
    const sig = raw ?? {};
    return {
        giant: normalizeGiant(sig.giant),
        rail: normalizeRail(sig.rail),
        crosshairs: normalizeCross(sig.crosshairs),
        accent: normalizeAccent(sig.accent),
        frame: sig.frame ?? null,
        cornerTicks: !!sig.cornerTicks,
        mastheadFamily: sig.mastheadFamily,
        mastheadWeight: sig.mastheadWeight,
        typeScale: sig.typeScale,
        spread: sig.spread ?? 'classic',
    };
};

// §8 · DECORATION EMITTERS — every device expressed in the house unit
//      grammar (x/y fractions, w/h percent), every anchor deliberate.
//      The renderer re-expands these into full-fidelity layers.

export interface DecorationShape {
    kind: 'frame' | 'crosshair' | 'accent-block' | 'em-dash' | 'giant-type' | 'rail-bg' | 'corner-tick';
    xPct: number;
    yPct: number;
    wPct: number;
    hPct: number;
    color: string;
    rotationDeg: number;
    depth: 'behind' | 'front';
    text?: string;
    typeFamily?: string;
    weight?: number;
    letterSpacing?: number;
    maskPhoto?: boolean;
    /** stroke weight for frames, PERCENT */
    thickness?: number;
    /** crosshair arm thickness, FRACTION of canvas */
    arm?: number;
    corner?: number;
    opacity?: number;
}

interface EmitContext {
    g: Grid;
    word: string;
    ladder: TypeLadder;
}

const emitDecorations = (sig: NormalizedSignature, ctx: EmitContext): DecorationShape[] => {
    const out: DecorationShape[] = [];
    const ink = ctx.g.light ? '#2B2B2B' : '#F5F0E8';

    // GIANT TYPE-AS-MASK WORDMARK — centre-anchored, photo reading through
    // the letterforms. The composer's masthead retires for giant styles
    // (the apply pass hides it): the wordmark IS the masthead.
    if (sig.giant) {
        const gs = sig.giant;
        out.push({
            kind: 'giant-type',
            xPct: 0.5,
            yPct: 0.5,
            wPct: gs.widthPct ?? 92,
            hPct: gs.heightPct ?? 90,
            color: ink,
            rotationDeg: 0,
            depth: 'behind',
            text: gs.word && gs.word.trim() ? gs.word.trim().toUpperCase() : ctx.word,
            typeFamily: sig.mastheadFamily ?? 'geometric-sans',
            weight: gs.weight ?? sig.mastheadWeight ?? 900,
            letterSpacing: gs.tracking ?? -0.03,
            maskPhoto: gs.maskPhoto ?? true,
            opacity: gs.opacity ?? 0.14,
        });
    }

    // THE RAIL — a right-edge editorial column backdrop with em-dash
    // dividers on a shared baseline grid. DECORATIVE ONLY by ruling: the
    // cluster's placement stays optimizer-owned; if the cost matrix sends
    // the cluster to the rail column (as rail styles' calm photos usually
    // will), the backdrop is already there waiting for it.
    if (sig.rail) {
        const rs = sig.rail;
        const rx = rs.x ?? 0.875;
        const rw = rs.widthPct ?? 18;
        if (rs.backdrop) {
            out.push({
                kind: 'rail-bg',
                xPct: rx,
                yPct: 0.5,
                wPct: rw,
                hPct: 78,
                color: ink,
                rotationDeg: 0,
                depth: 'behind',
                opacity: 0.08,
            });
        }
        if (rs.dividers !== false) {
            const top = 0.14;
            const bottom = 0.86;
            const bands = 4;
            const step = (bottom - top) / bands;
            for (let i = 1; i < bands; i++) {
                out.push({
                    kind: 'em-dash',
                    xPct: rx,
                    yPct: Math.round((top + step * i) * 1000) / 1000,
                    wPct: Math.min(7.2, rw * 0.4),
                    hPct: 0.4,
                    color: ink,
                    rotationDeg: 0,
                    depth: 'front',
                    thickness: 0.18,
                });
            }
        }
    }

    // CROSSHAIR REGISTRATION MARKS — the print-shop's fingerprint.
    if (sig.crosshairs) {
        const cs = sig.crosshairs;
        const positions = cs.positions ?? ([[0.08, 0.08], [0.92, 0.92]] as [number, number][]);
        for (const [chx, chy] of positions) {
            out.push({
                kind: 'crosshair',
                xPct: chx,
                yPct: chy,
                wPct: cs.sizePct ?? 3,
                hPct: cs.sizePct ?? 3,
                color: ink,
                rotationDeg: 0,
                depth: 'front',
                arm: cs.arm ?? 0.018,
            });
        }
    }

    // ACCENT BLOCK — one hue of the house, placed with intent.
    if (sig.accent) {
        const a = sig.accent;
        out.push({
            kind: 'accent-block',
            xPct: a.xPct ?? 0.24,
            yPct: a.yPct ?? 0.5,
            wPct: a.wPct ?? 12,
            hPct: a.hPct ?? 8,
            color: a.color,
            rotationDeg: 0,
            depth: 'behind',
        });
    }

    // THE BORDER FRAME — turns the cover into a document.
    if (sig.frame) {
        const inset = sig.frame.inset ?? 0.035;
        out.push({
            kind: 'frame',
            xPct: 0.5,
            yPct: 0.5,
            wPct: Math.round((1 - inset * 2) * 100),
            hPct: Math.round((1 - inset * 2) * 94),
            color: sig.frame.color,
            rotationDeg: 0,
            depth: 'front',
            thickness: sig.frame.strokeW,
        });
    }

    // (Corner-tick crop marks were emitted here once and REJECTED by
    // the editor — nothing is ever placed at the four corners. The
    // "add relevant details" Gestalt fix is carried by REAL editorial
    // furniture only: dateline, scan, badges, rail dividers.)

    return out;
};

// ===========================================================================
// §9 · THE LAW VERIFIER — the engine's conscience. Eleven laws, checked
//      against the FINAL geometry; survivors report home with severity so
//      the UI can show exactly what the mathematics could not give.
//      L9–L11 are the GESTALT calibration (Aevy lesson): proximity,
//      alignment/continuity, and focal contrast — measured, not assumed.
// ===========================================================================

export interface LayoutViolation {
    id: string;
    severity: 'critical' | 'warning' | 'info';
    message: string;
    role?: string;
}

const verifyLaws = (
    placements: SolvedPlacement[],
    ladder: TypeLadder,
    g: Grid,
    sig: NormalizedSignature,
): LayoutViolation[] => {
    const v: LayoutViolation[] = [];
    const byRole: Record<string, SolvedPlacement> = {};
    for (const p of placements) byRole[p.role] = p;

    // L1 · MASTHEAD LEADERSHIP — the brand tops the upper field (giant
    // wordmark styles are exempt: the wordmark IS the masthead).
    const mast = byRole.masthead;
    if (mast && mast.yPct > 0.55 && !sig.giant) {
        v.push({ id: 'masthead-lead', severity: 'critical', message: 'masthead must lead the upper field', role: 'masthead' });
    }

    // L2 · LADDER MONOTONICITY — every rung steps down, visibly.
    if (!(ladder.masthead >= ladder.feature * 2.0)) {
        v.push({ id: 'ladder-monotonic', severity: 'critical', message: 'masthead rung must tower the feature ≥ 2×' });
    }
    if (!(ladder.feature >= ladder.sub * 1.4 && ladder.sub >= ladder.teaser * 1.1)) {
        v.push({ id: 'ladder-steps', severity: 'warning', message: 'feature/sub/teaser rungs must step down clearly' });
    }

    // L3 · CLUSTER BREATH — the sub lands a full breath beneath the feature.
    const feat = byRole.feature;
    const sub = byRole.sub;
    if (feat && sub && feat.heightPct && sub.heightPct) {
        const need = feat.heightPct / 200 + ladder.breath + sub.heightPct / 200;
        if (sub.yPct - feat.yPct < need * 0.85) {
            v.push({ id: 'cluster-breath', severity: 'warning', message: 'sub must breathe one gap beneath the feature', role: 'sub' });
        }
    }

    // L4 · EDGE SAFETY — nobody hangs off the page (rails exempt by job).
    for (const p of placements) {
        if (p.rotationDeg) continue;
        const hw = p.widthPct / 200;
        if (p.xPct - hw < 0.015 || p.xPct + hw > 0.985) {
            v.push({ id: 'edge-safety', severity: 'warning', message: `${p.role} crosses the page margin`, role: p.role });
        }
    }

    // L5 · PLATE JUSTIFICATION — plates only where physics demanded them.
    if (feat && feat.plate && feat.zoneLum != null) {
        const cr = contrastRatio(g.light ? PAPER_LUM : INK_LUM, feat.zoneLum);
        if (cr >= 4.5) {
            v.push({ id: 'plate-justified', severity: 'info', message: 'feature plated on already-contrasting ground', role: 'feature' });
        }
    }

    // L6 · SCAN CORNER — the barcode/QR sits inside a real corner.
    const scan = byRole.scan;
    if (scan && (scan.xPct < 0.05 || scan.xPct > 0.95 || scan.yPct < 0.05 || scan.yPct > 0.95)) {
        v.push({ id: 'scan-corner', severity: 'warning', message: 'scan element must sit inside a corner', role: 'scan' });
    }

    // L7 · SPREAD BALANCE — the page breathes in more than one quadrant.
    const m = [0, 0, 0, 0];
    for (const p of placements) {
        const w = p.role === 'masthead' ? 2 : p.role === 'feature' ? 3 : 1;
        m[quadrantOf(p.xPct, p.yPct)] += w;
    }
    const total = m.reduce((a, b) => a + b, 0) || 1;
    if (Math.max(...m) / total > 0.75) {
        v.push({ id: 'spread-balance', severity: 'warning', message: 'text mass crowds one quadrant — the page cannot breathe' });
    }

    // L8 · RAIL ALIGNMENT — with a rail backdrop, the cluster honours it.
    if (sig.rail) {
        const rx = sig.rail.x ?? 0.875;
        if (feat && Math.abs(feat.xPct - rx) > 0.14) {
            v.push({ id: 'rail-alignment', severity: 'info', message: 'rail present but the cluster chose its own column', role: 'feature' });
        }
    }

    // L9 · GESTALT PROXIMITY — "keep related elements together so they
    // appear part of the same group." The cluster is ONE conceptual
    // unit; the masthead is ANOTHER. The void between groups must be
    // clearly larger than the breaths inside the cluster — a masthead
    // parked inside the cluster band merges the two groups and the
    // hierarchy dissolves. (Rotated rail mastheads are exempt: their
    // band math is columnar, not horizontal.)
    const teas = byRole.teasers;
    if (feat && sub && mast && !mast.rotationDeg) {
        const mastBottom = mast.yPct + ladder.masthead / 200;
        const clTopY = Math.min(
            feat.yPct - (feat.heightPct ?? 0) / 200,
            sub.yPct - (sub.heightPct ?? 0) / 200,
            teas ? teas.yPct - (teas.heightPct ?? 0) / 200 : feat.yPct,
        );
        const between = clTopY - mastBottom;
        const xOverlap =
            Math.abs(mast.xPct - feat.xPct) < ((mast.widthPct + feat.widthPct) / 200) * 0.5;
        if (between >= 0 && between < ladder.breath * 2 && xOverlap) {
            v.push({
                id: 'gestalt-proximity',
                severity: 'warning',
                message: 'masthead sits inside the cluster band — groups must separate harder than their internal breaths',
                role: 'masthead',
            });
        }
    }

    // L10 · GESTALT ALIGNMENT / CONTINUITY — "arrange elements in a
    // common direction so the viewer's eye moves naturally." The
    // cluster's rungs share ONE axis; the eye rides it down the page.
    // Drift off the axis (beyond optical tolerance) breaks the path.
    if (feat && sub && teas) {
        const axisSpread = Math.max(
            Math.abs(feat.xPct - sub.xPct),
            Math.abs(sub.xPct - teas.xPct),
            Math.abs(feat.xPct - teas.xPct),
        );
        if (axisSpread > 0.02) {
            v.push({
                id: 'gestalt-alignment',
                severity: 'warning',
                message: 'cluster rungs drift off the shared axis — the eye loses its path',
                role: 'feature',
            });
        }
    }

    // L11 · GESTALT FOCAL POINT — "no focal point? use contrast." The
    // feature owns scale dominance (L2); this law owns the TONAL side:
    // a headline neither plated nor sitting on ≥2.5:1 ground has no
    // focal contrast at all — the cover becomes wallpaper.
    if (feat && feat.zoneLum != null && !feat.plate) {
        const cr = contrastRatio(g.light ? PAPER_LUM : INK_LUM, feat.zoneLum);
        if (cr < 2.5) {
            v.push({
                id: 'gestalt-focal',
                severity: 'critical',
                message: 'feature has neither a plate nor tonal contrast — no focal point',
                role: 'feature',
            });
        }
    }

    return v;
};

/** Human-readable one-liner for the ⚙ SOLVER readout. */
export const formatSolverReport = (s: SolverResult): string =>
    `${String(s.style).toUpperCase()} · ${s.placements.length} placements · ${s.decorations.length} decorations · ` +
    `${s.violations.length} law flags · +${s.alternates.length} alternates · seed ${s.seed.toString(36)}`;

// ===========================================================================
// §10 · APPLICATION — the Gen II contract, unchanged: solved placements
//       rewrite the composed cover's specs; the app's layout-law engine
//       verifies physically afterwards. Field writes go through a typed
//       bridge so this file compiles against any composer generation.
// ===========================================================================

const roleOfSpec = (spec: { id: string }): string | null => {
    const id = spec.id;
    if (id.includes('masthead')) return 'masthead';
    if (id.includes('dateline') || id.includes('rail')) return 'dateline';
    if (id.includes('feature')) return 'feature';
    if (id.includes('sub')) return 'sub';
    if (id.includes('teaser')) return 'teasers';
    if (id.includes('badge')) return 'badge';
    return null;
};

/** Writable bridge — the solver's word is law on these fields. */
type WritableText = Pick<SystemTextSpec, 'xPct' | 'yPct' | 'widthPct' | 'heightPct' | 'rotationDeg' | 'depth'> & {
    opacity?: number;
};
const asWritable = (s: SystemTextSpec): WritableText => s as unknown as WritableText;

/** Flat-rect approximation of a decoration for the composed cover's shape
 *  list. The app's renderer re-expands every `sys-decoration-` id with
 *  full fidelity (giant type becomes a real text layer, crosshairs become
 *  twin registration bars), so this only guarantees the shapes exist. */
const decorationToShapeSpec = (d: DecorationShape): SystemShapeSpec =>
    ({
        id: `sys-decoration-${d.kind}-${Math.round(d.xPct * 1000)}-${Math.round(d.yPct * 1000)}`,
        kind: 'rect',
        xPct: d.xPct,
        yPct: d.yPct,
        wPct: d.wPct,
        hPct: d.hPct,
        fillColor: d.color,
        fillOn: d.kind !== 'frame',
        strokeColor: d.color,
        strokeW: d.kind === 'frame' ? (d.thickness ?? 1.5) : 0,
        cornerPct: d.corner ?? 0,
        rotationDeg: d.rotationDeg,
        opacity: d.opacity ?? 1,
        depth: d.depth,
    }) as SystemShapeSpec;

/** Mutates the composed cover in place: every text spec snaps to its solved
 *  zone with its solved rung — and the preset's AUTHORED relationships ride
 *  along (siblings keep their designed offsets from their role leader).
 *  A weighted relaxation pass then guarantees the nothing-collides contract
 *  at spec level, BEFORE any renderer draws a pixel. The feature gains a
 *  plate when the physics demands one; the barcode/QR moves to the solved
 *  corner; the style signature's decorations land as shapes. The layout
 *  laws verify after. */
export function applySolverToCover(cover: ComposedCover, solved: SolverResult): void {
    const byRole: Record<string, SolvedPlacement> = {};
    for (const p of solved.placements) byRole[p.role] = p;

    // ── SKELETON SNAPSHOT ── before any coordinate is touched, record
    // each role's LEADER spec and every sibling's designed offset from
    // it. The presets are authored arrangements, not noise: when the
    // leader moves to its solved zone, the siblings ride along with
    // their relationships intact. (The editor's ruling: stop fighting
    // the designed layouts — fit them.)
    const specs = cover.text as SystemTextSpec[];
    const leaderOf = new Map<string, SystemTextSpec>();
    const siblingDelta: Array<{ spec: SystemTextSpec; role: string; dx: number; dy: number }> = [];
    for (const spec of specs) {
        const role = roleOfSpec(spec);
        if (!role) continue;
        const lead = leaderOf.get(role);
        if (!lead) leaderOf.set(role, spec);
        else siblingDelta.push({ spec, role, dx: spec.xPct - lead.xPct, dy: spec.yPct - lead.yPct });
    }

    const used = new Set<string>();
    for (const spec of specs) {
        const role = roleOfSpec(spec);
        if (!role) continue;
        const p = byRole[role];
        if (!p || used.has(role)) continue; // first spec of each role carries the zone
        used.add(role);
        const w = asWritable(spec);
        w.xPct = p.xPct;
        w.yPct = p.yPct;
        w.widthPct = p.widthPct;
        if (p.heightPct) w.heightPct = p.heightPct; // the solved rung — the ladder speaks
        if (p.rotationDeg) w.rotationDeg = p.rotationDeg;
        if (p.depth) w.depth = p.depth;
    }

    // Siblings re-dock to their (now moved) leader — designed spacing,
    // translated as one arrangement.
    for (const s of siblingDelta) {
        const lead = leaderOf.get(s.role);
        if (!lead) continue;
        const w = asWritable(s.spec);
        w.xPct = clamp(lead.xPct + s.dx, 0.03, 0.97);
        w.yPct = clamp(lead.yPct + s.dy, 0.03, 0.97);
    }

    // ── §10.5 · THE RELAXATION PASS — the nothing-collides contract ──
    // Every live front text spec is checked pairwise; intersecting pairs
    // are pushed apart VERTICALLY (so every shared alignment axis — the
    // Gestalt continuity law — survives the push), split by inverse role
    // mass: the masthead barely moves, a teaser yields completely.
    // ≤10 relaxation passes over a handful of specs; then a final audit
    // that must come back clean or report itself as critical. This is
    // where "things appear on other things" dies — at the spec, before
    // any renderer ever draws.
    const MASS: Record<string, number> = { masthead: 5, feature: 4, sub: 2.5, teasers: 1.2, dateline: 1.5, badge: 1 };
    const boxOf = (s: SystemTextSpec): Rect => ({
        x0: s.xPct - s.widthPct / 200,
        y0: s.yPct - (s.heightPct ?? 3) / 200,
        x1: s.xPct + s.widthPct / 200,
        y1: s.yPct + (s.heightPct ?? 3) / 200,
    });
    const live = specs.filter((s) => !(s as { opacity?: number }).opacity && !s.rotationDeg);
    for (let pass = 0; pass < 10; pass++) {
        let moved = false;
        for (let i = 0; i < live.length; i++) {
            for (let j = i + 1; j < live.length; j++) {
                const a = live[i];
                const b = live[j];
                const hit = rectIntersect(boxOf(a), boxOf(b));
                if (!hit) continue;
                const ma = MASS[roleOfSpec(a) ?? ''] ?? 1;
                const mb = MASS[roleOfSpec(b) ?? ''] ?? 1;
                const pushY = hit.y1 - hit.y0 + 0.012; // full separation plus one breath
                const dir = a.yPct <= b.yPct ? -1 : 1; // upper spec rises, lower sinks
                const total = ma + mb;
                const wa = asWritable(a);
                const wb = asWritable(b);
                wa.yPct = clamp(wa.yPct + dir * pushY * (mb / total), 0.04, 0.96);
                wb.yPct = clamp(wb.yPct - dir * pushY * (ma / total), 0.04, 0.96);
                moved = true;
            }
        }
        if (!moved) break;
    }
    // The audit — the relaxation must end clean or say so, loudly.
    solved.violations = solved.violations ?? [];
    for (let i = 0; i < live.length; i++) {
        for (let j = i + 1; j < live.length; j++) {
            if (rectIntersect(boxOf(live[i]), boxOf(live[j]))) {
                solved.violations.push({
                    id: 'spec-overlap',
                    severity: 'critical',
                    message: `${roleOfSpec(live[i]) ?? 'text'} still intersects ${roleOfSpec(live[j]) ?? 'text'} after relaxation`,
                    role: roleOfSpec(live[i]) ?? undefined,
                });
            }
        }
    }

    // Plate the feature when the solver's physics demands one.
    const featP = byRole.feature;
    const featSpec = (cover.text as SystemTextSpec[]).find((s) => roleOfSpec(s) === 'feature');
    if (featP?.plate && featSpec) {
        const hasPlate = (cover.shapes as SystemShapeSpec[]).some(
            (s) => s.kind === 'rect' && s.fillOn && Math.abs(s.xPct - featSpec.xPct) < 0.12 && Math.abs(s.yPct - featSpec.yPct) < 0.12,
        );
        if (!hasPlate) {
            const w = featSpec.widthPct + 8;
            const h = featSpec.heightPct + 2.5;
            (cover.shapes as SystemShapeSpec[]).push({
                id: 'sys-solver-plate-0',
                kind: 'rect',
                xPct: featSpec.xPct,
                yPct: featSpec.yPct,
                wPct: w,
                hPct: h,
                fillColor: solved.photoLight ? '#000000' : '#FFFFFF',
                fillOn: true,
                strokeColor: solved.photoLight ? '#000000' : '#FFFFFF',
                strokeW: 0,
                cornerPct: 0,
                rotationDeg: 0,
                opacity: 1,
                depth: 'behind',
                // Plates render under front text; the civic-sense render guard
                // flips the text ink against whatever sits beneath it.
            } as SystemShapeSpec);
        }
    }

    // The scan element moves to the solved corner.
    const scanP = byRole.scan;
    if (scanP) {
        const sc = (cover.shapes as SystemShapeSpec[]).find((s) => s.kind === 'barcode' || s.kind === 'qr');
        if (sc) {
            sc.xPct = scanP.xPct;
            sc.yPct = scanP.yPct;
        }
    }

    // Giant wordmark styles retire the composer's masthead — printing both
    // would set the word twice.
    const sig = resolveSignature(solved.style);
    if (sig.giant) {
        const m = (cover.text as SystemTextSpec[]).find((s) => roleOfSpec(s) === 'masthead');
        if (m) asWritable(m).opacity = 0;
    }

    for (const d of solved.decorations) {
        (cover.shapes as SystemShapeSpec[]).push(decorationToShapeSpec(d));
    }
}

// §11 · APPENDIX A — ADDING A PRESET (the two-minute guide)
// --------------------------------------------------------------------------
// A preset is a NAME plus a StyleSignature. Nothing else. The engine's
// optimizer, ladder, physics and verifier apply to it like any genre.
//
//   1. Register (once, at app boot or module init):
//
//        registerStyleSignature('preset:fragments', {
//            giant: { heightPct: 88, widthPct: 74, maskPhoto: true },
//            rail: { x: 0.875, widthPct: 18, dividers: true },
//            crosshairs: { sizePct: 3 },
//            accent: '#D94B28',
//        });
//
//   2. Solve with it:
//
//        solveEditorialCover(grid, semantics, { style: 'preset:fragments' });
//
//   3. Devices available (all optional, all composable):
//        giant       boolean | GiantSpec   — type-as-mask wordmark
//        rail        boolean | RailSpec    — edge column + em-dash dividers
//        crosshairs  boolean | CrosshairSpec — registration marks
//        accent      string  | AccentSpec  — one hue of the house
//        frame       FrameSpec              — border document frame
//        cornerTicks boolean                — reserved, never emitted (corner marks rejected)
//        typeScale   Partial<Record<SolverRole, number>> — rung overrides
//        spread      'tight' | 'classic' | 'wide'
//
//   THE RULING, ENCODED: signatures decorate, the optimizer decides. A
//   preset can never seize geometry — the flat-rail era cannot recur.
//
// ===========================================================================
// §12 · APPENDIX B — PERFORMANCE BUDGET (why this stays instant)
// --------------------------------------------------------------------------
//   analyzeGridFromCanvas : 1 downsample + 1 thumbnail read + O(96×120)
//                           pixel pass + O(720) cell passes + a BFS whose
//                           queue is bounded by 720 cells. Sub-millisecond.
//   enumerateZones        : ≤ 40 growth seeds × O(54) growth + NMS over
//                           ≤ 80 candidates at O(k²) with k ≤ 80 → ≤ 6.4k
//                           IoU checks of pure arithmetic.
//   hungarian             : 5×5 padded matrix, ~125 inner iterations a
//                           solve, run ≤ 4 + 2 times for k-best + rebalance.
//   solveEditorialCover   : all of the above + seven small geometry loops.
//                           Comfortably under 1 ms on anything modern.
//
//   Same photo + same seed → byte-identical cover. Always.
